FPS 컨트롤러 개요

1인칭 가장 기본적인 3D 캐릭터 컨트롤러

씬 트리 구조

# FPS 컨트롤러 노드 구조
# Player (CharacterBody3D)
#  +-- CollisionShape3D (캡슐)
#  +-- CameraPivot (Node3D)  -- 상하 회전용
#  |   +-- Camera3D           -- 눈 위치
#  +-- RayCast3D              -- 바닥 감지
#  +-- MeshInstance3D         -- 디버그용 몸체
Tip: CameraPivot을 캐릭터 머리 높이(약 1.6m)에 배치하면 자연스러운 눈높이가 됩니다. CollisionShape3D는 높이 1.8m, 반지름 0.3m 캡슐이 일반적입니다.

핵심 구성 요소

  • CharacterBody3D - move_and_slide() 기반 이동
  • CollisionShape3D - 캡슐 충돌체 (CapsuleShape3D)
  • CameraPivot - X축 회전 (상하 룩)
  • Player 루트 - Y축 회전 (좌우 룩)
  • Camera3D - current = true로 활성화
주의: Camera3D를 직접 Player에 붙이면 상하좌우 회전이 섞여 짐벌 락 문제가 발생합니다. 반드시 피봇 구조를 사용하세요.

마우스 캡처 (Input.MOUSE_MODE_CAPTURED)

커서를 숨기고 화면 중앙에 고정

extends CharacterBody3D

func _ready():
    # 마우스를 캡처하여 화면 중앙에 고정
    Input.set_mouse_mode(Input.MOUSE_MODE_CAPTURED)

func _unhandled_input(event: InputEvent):
    # ESC로 마우스 해제
    if event.is_action_pressed("ui_cancel"):
        Input.set_mouse_mode(Input.MOUSE_MODE_VISIBLE)
    # 마우스 클릭으로 다시 캡처
    if event is InputEventMouseButton and event.pressed:
        if Input.get_mouse_mode() == Input.MOUSE_MODE_VISIBLE:
            Input.set_mouse_mode(Input.MOUSE_MODE_CAPTURED)
마우스 모드 커서 표시 커서 이동 용도
MOUSE_MODE_VISIBLE 보임 자유 UI, 메뉴
MOUSE_MODE_CAPTURED 숨김 중앙 고정 FPS/TPS 게임플레이
MOUSE_MODE_HIDDEN 숨김 자유 커스텀 커서
MOUSE_MODE_CONFINED 보임 창 내부 제한 RTS, 전략 게임
Tip: 일시정지 메뉴를 열 때 MOUSE_MODE_VISIBLE로, 닫을 때 MOUSE_MODE_CAPTURED로 전환하면 자연스러운 흐름이 됩니다.

마우스 룩 (회전)

핵심 마우스 상대 이동량으로 카메라 회전

@onready var camera_pivot: Node3D = $CameraPivot

@export var mouse_sensitivity: float = 0.002  # 마우스 감도

func _unhandled_input(event: InputEvent):
    if event is InputEventMouseMotion:
        if Input.get_mouse_mode() == Input.MOUSE_MODE_CAPTURED:
            # 좌우 회전: 플레이어 루트를 Y축으로 회전
            rotate_y(-event.relative.x * mouse_sensitivity)

            # 상하 회전: 카메라 피봇을 X축으로 회전
            camera_pivot.rotate_x(-event.relative.y * mouse_sensitivity)

            # 상하 회전 각도 제한 (-90도 ~ +90도)
            camera_pivot.rotation.x = clamp(
                camera_pivot.rotation.x,
                deg_to_rad(-89),  # 아래 한계
                deg_to_rad(89)    # 위 한계
            )

마우스 룩 핵심 원리

  • event.relative - 이전 프레임 대비 마우스 이동 픽셀 (Vector2)
  • rotate_y() - 플레이어 루트의 수평 회전 (좌우 둘러보기)
  • rotate_x() - 카메라 피봇의 수직 회전 (위아래 둘러보기)
  • clamp() - 수직 회전을 -89도~+89도로 제한하여 뒤집힘 방지
주의: 감도 값이 너무 크면 어지러움을 유발합니다. 0.001~0.005 범위에서 시작하고 옵션 메뉴에서 사용자가 조절할 수 있게 만드세요.

카메라 피봇 구조

수평/수직 회전을 분리하는 핵심 구조

FPS 피봇

# FPS: 피봇이 머리 높이에 위치
# Player (CharacterBody3D)
#   rotation.y = 좌우 회전
#   +-- CameraPivot (Node3D)
#       position.y = 1.6  (눈 높이)
#       rotation.x = 상하 회전
#       +-- Camera3D
#           position = (0, 0, 0)

# 에디터에서 설정:
# CameraPivot.position = Vector3(0, 1.6, 0)
# Camera3D.current = true

TPS 피봇

# TPS: 카메라가 뒤로 떨어져 있음
# Player (CharacterBody3D)
#   rotation.y = 좌우 회전
#   +-- CameraPivot (Node3D)
#       position.y = 1.6
#       rotation.x = 상하 회전
#       +-- SpringArm3D
#           spring_length = 4.0
#           +-- Camera3D
#               position = (0, 0, 0)

# SpringArm3D가 벽 충돌 시
# 자동으로 카메라를 당겨줌

피봇 구조의 장점

  • 축 분리 - Y축 회전(루트)과 X축 회전(피봇)이 독립적
  • 짐벌 락 방지 - 두 축이 섞이지 않아 안정적
  • FPS/TPS 전환 용이 - 카메라 위치만 바꾸면 전환 가능
  • 추가 효과 - 카메라 흔들림, 기울기 등을 피봇에 적용 가능

WASD 이동 (방향 벡터 변환)

핵심 입력 방향을 카메라 기준으로 변환

@export var speed: float = 5.0

func _physics_process(delta: float):
    # 1. 입력 벡터 구하기 (정규화됨)
    var input_dir := Input.get_vector("move_left", "move_right", "move_forward", "move_back")

    # 2. 플레이어의 바라보는 방향 기준으로 변환
    #    transform.basis.z = 로컬 후방 벡터 (Godot은 -Z가 전방)
    #    transform.basis.x = 로컬 우측 벡터
    var direction := (transform.basis * Vector3(input_dir.x, 0, input_dir.y)).normalized()

    # 3. 수평 속도 적용
    if direction:
        velocity.x = direction.x * speed
        velocity.z = direction.z * speed
    else:
        velocity.x = 0.0
        velocity.z = 0.0

    move_and_slide()

방향 벡터 변환 원리

  • Input.get_vector() - 4방향 입력을 Vector2(-1~1, -1~1)로 반환
  • transform.basis - 노드의 로컬 좌표축 (3x3 행렬)
  • basis * Vector3(x, 0, y) - 로컬 좌표를 월드 좌표로 변환
  • .normalized() - 대각선 이동 시 속도 초과 방지
Tip: 프로젝트 설정에서 move_forward, move_back, move_left, move_right 입력 액션을 W, S, A, D 키로 매핑하세요.

이동 + 마우스 룩 통합

FPS 컨트롤러의 기본 뼈대 완성

extends CharacterBody3D

@onready var camera_pivot: Node3D = $CameraPivot

@export var speed: float = 5.0
@export var mouse_sensitivity: float = 0.002

func _ready():
    Input.set_mouse_mode(Input.MOUSE_MODE_CAPTURED)

func _unhandled_input(event: InputEvent):
    # 마우스 룩
    if event is InputEventMouseMotion:
        if Input.get_mouse_mode() == Input.MOUSE_MODE_CAPTURED:
            rotate_y(-event.relative.x * mouse_sensitivity)
            camera_pivot.rotate_x(-event.relative.y * mouse_sensitivity)
            camera_pivot.rotation.x = clamp(
                camera_pivot.rotation.x, deg_to_rad(-89), deg_to_rad(89)
            )
    # ESC 토글
    if event.is_action_pressed("ui_cancel"):
        Input.set_mouse_mode(Input.MOUSE_MODE_VISIBLE)
    elif event is InputEventMouseButton and event.pressed:
        if Input.get_mouse_mode() == Input.MOUSE_MODE_VISIBLE:
            Input.set_mouse_mode(Input.MOUSE_MODE_CAPTURED)

func _physics_process(delta: float):
    # 이동
    var input_dir := Input.get_vector("move_left", "move_right", "move_forward", "move_back")
    var direction := (transform.basis * Vector3(input_dir.x, 0, input_dir.y)).normalized()

    if direction:
        velocity.x = direction.x * speed
        velocity.z = direction.z * speed
    else:
        velocity.x = 0.0
        velocity.z = 0.0

    move_and_slide()

중력과 점프 (3D)

물리 자연스러운 낙하와 점프 구현

@export var jump_velocity: float = 4.5
@export var coyote_time: float = 0.15  # 코요테 타임

# 프로젝트 설정에서 중력 값 가져오기
var gravity: float = ProjectSettings.get_setting("physics/3d/default_gravity")
var coyote_timer: float = 0.0

func _physics_process(delta: float):
    # 중력 적용
    if not is_on_floor():
        velocity.y -= gravity * delta
        coyote_timer -= delta
    else:
        coyote_timer = coyote_time  # 바닥이면 코요테 타이머 리셋

    # 점프 (바닥 또는 코요테 타임 내)
    if Input.is_action_just_pressed("jump"):
        if is_on_floor() or coyote_timer > 0.0:
            velocity.y = jump_velocity
            coyote_timer = 0.0  # 이중 점프 방지

    # ... 이동 코드 ...
    move_and_slide()

중력 관련 값

  • 기본 중력 - 9.8 m/s (프로젝트 설정)
  • 점프 속도 - 4.5 ~ 6.0 (게임에 따라 조절)
  • 코요테 타임 - 0.1 ~ 0.2초 권장
Tip: 더 자연스러운 점프를 위해 상승 시에는 기본 중력을, 하강 시에는 1.5~2배 중력을 적용하면 게임이 한결 쾌적해집니다. if velocity.y < 0: velocity.y -= gravity * 1.5 * delta

가속/감속 이동

부드러운 움직임을 위한 보간 적용

lerp 방식 (부드러운)

@export var speed: float = 5.0
@export var acceleration: float = 10.0
@export var deceleration: float = 8.0
@export var air_control: float = 0.3

func _physics_process(delta: float):
    var input_dir := Input.get_vector(
        "move_left", "move_right",
        "move_forward", "move_back"
    )
    var direction := (transform.basis
        * Vector3(input_dir.x, 0, input_dir.y)
    ).normalized()

    var accel = acceleration if is_on_floor() \
        else acceleration * air_control

    if direction:
        velocity.x = lerp(
            velocity.x,
            direction.x * speed,
            accel * delta
        )
        velocity.z = lerp(
            velocity.z,
            direction.z * speed,
            accel * delta
        )
    else:
        velocity.x = lerp(
            velocity.x, 0.0,
            deceleration * delta
        )
        velocity.z = lerp(
            velocity.z, 0.0,
            deceleration * delta
        )

move_toward 방식 (일정한)

@export var speed: float = 5.0
@export var accel_rate: float = 50.0
@export var decel_rate: float = 40.0

func apply_movement(
    direction: Vector3, delta: float
):
    var target_vel := direction * speed
    var current_vel := Vector3(
        velocity.x, 0, velocity.z
    )

    if direction:
        current_vel = current_vel.move_toward(
            target_vel,
            accel_rate * delta
        )
    else:
        current_vel = current_vel.move_toward(
            Vector3.ZERO,
            decel_rate * delta
        )

    velocity.x = current_vel.x
    velocity.z = current_vel.z

비교

  • lerp - 지수적 감속, 부드럽지만 완전히 0에 도달하지 않음
  • move_toward - 선형 감속, 예측 가능하고 정확히 0에 도달

스프린트

기능 달리기 속도 + 스태미나 + FOV 변화

@export var walk_speed: float = 5.0
@export var sprint_speed: float = 8.5
@export var sprint_fov: float = 85.0
@export var normal_fov: float = 75.0
@export var max_stamina: float = 100.0
@export var stamina_drain: float = 20.0   # 초당 소모
@export var stamina_regen: float = 15.0   # 초당 회복

@onready var camera: Camera3D = $CameraPivot/Camera3D

var current_stamina: float
var is_sprinting: bool = false

func _ready():
    current_stamina = max_stamina

func _physics_process(delta: float):
    # 스프린트 판정
    is_sprinting = (Input.is_action_pressed("sprint")
        and current_stamina > 0.0
        and Input.get_vector("move_left", "move_right",
            "move_forward", "move_back").length() > 0.1)

    # 속도 결정
    var current_speed = sprint_speed if is_sprinting else walk_speed

    # 스태미나 관리
    if is_sprinting:
        current_stamina -= stamina_drain * delta
        current_stamina = max(0.0, current_stamina)
    else:
        current_stamina += stamina_regen * delta
        current_stamina = min(max_stamina, current_stamina)

    # FOV 부드러운 전환
    var target_fov = sprint_fov if is_sprinting else normal_fov
    camera.fov = lerp(camera.fov, target_fov, 5.0 * delta)

    # ... 이동 코드에서 current_speed 사용 ...
Tip: 스태미나가 0이 되면 일정 시간(예: 2초) 스프린트를 사용하지 못하게 쿨다운을 추가하면 무한 토글을 방지할 수 있습니다.

웅크리기 (Crouch)

기능 충돌 형태 축소 + 카메라 높이 변경

@export var stand_height: float = 1.8
@export var crouch_height: float = 1.0
@export var stand_camera_y: float = 1.6
@export var crouch_camera_y: float = 0.8
@export var crouch_speed: float = 2.5

@onready var collision: CollisionShape3D = $CollisionShape3D
@onready var camera_pivot: Node3D = $CameraPivot
@onready var head_check: RayCast3D = $HeadCheck  # 위쪽으로 쏘는 레이

var is_crouching: bool = false

func _physics_process(delta: float):
    # 웅크리기 입력
    var wants_crouch = Input.is_action_pressed("crouch")

    if wants_crouch and not is_crouching:
        start_crouch()
    elif not wants_crouch and is_crouching:
        # 머리 위에 공간이 있는지 확인
        if not head_check.is_colliding():
            end_crouch()

    # 카메라 높이 부드러운 전환
    var target_y = crouch_camera_y if is_crouching else stand_camera_y
    camera_pivot.position.y = lerp(camera_pivot.position.y, target_y, 10.0 * delta)

func start_crouch():
    is_crouching = true
    var shape = collision.shape as CapsuleShape3D
    shape.height = crouch_height
    collision.position.y = crouch_height / 2.0

func end_crouch():
    is_crouching = false
    var shape = collision.shape as CapsuleShape3D
    shape.height = stand_height
    collision.position.y = stand_height / 2.0
주의: HeadCheck RayCast3D는 위쪽(Vector3.UP)을 향하게 설정하고 target_position을 (0, stand_height - crouch_height, 0) 정도로 지정합니다. 반드시 enabled = true로 설정하세요.

TPS 컨트롤러 개요

3인칭 캐릭터 뒤에서 따라가는 카메라

TPS 씬 트리

# Player (CharacterBody3D)
#  +-- CollisionShape3D
#  +-- PlayerModel (Node3D)   -- 메시+애니
#  |   +-- AnimationPlayer
#  |   +-- MeshInstance3D / 임포트 모델
#  +-- CameraPivot (Node3D)
#  |   +-- SpringArm3D
#  |       spring_length = 4.0
#  |       +-- Camera3D
#  +-- RayCast3D
Tip: PlayerModel을 별도 Node3D로 감싸면, 카메라 회전과 캐릭터 모델 회전을 독립적으로 제어할 수 있습니다.

FPS vs TPS 비교

항목 FPS TPS
카메라 위치 눈 높이 뒤+위
캐릭터 모델 팔만 보임 전신 보임
캐릭터 회전 카메라와 동일 이동 방향으로 회전
벽 충돌 없음 SpringArm3D 필요
시야 좁음 (몰입감) 넓음 (공간 인식)

SpringArm3D 카메라 충돌 회피

벽이나 오브젝트 사이에서 자동으로 카메라 거리 조절

@onready var spring_arm: SpringArm3D = $CameraPivot/SpringArm3D

func _ready():
    Input.set_mouse_mode(Input.MOUSE_MODE_CAPTURED)

    # SpringArm3D 설정
    spring_arm.spring_length = 4.0     # 기본 카메라 거리
    spring_arm.collision_mask = 1       # 충돌 감지할 레이어
    spring_arm.margin = 0.2            # 벽에서의 여유 거리

    # SpringArm3D는 위치가 아닌 길이가 줄어듦
    # 장애물이 없으면 spring_length 유지
    # 장애물이 있으면 자동으로 줄어듦

줌 인/아웃

@export var min_zoom: float = 1.5
@export var max_zoom: float = 8.0
@export var zoom_step: float = 0.5

func _unhandled_input(event: InputEvent):
    if event is InputEventMouseButton:
        # 마우스 휠로 줌
        if event.button_index == MOUSE_BUTTON_WHEEL_UP:
            spring_arm.spring_length = max(
                min_zoom,
                spring_arm.spring_length - zoom_step
            )
        elif event.button_index == MOUSE_BUTTON_WHEEL_DOWN:
            spring_arm.spring_length = min(
                max_zoom,
                spring_arm.spring_length + zoom_step
            )

SpringArm3D 주요 속성

  • spring_length - 최대 팔 길이 (카메라 거리)
  • collision_mask - 감지할 충돌 레이어
  • margin - 충돌 시 여유 거리
  • shape - 레이 대신 형태로 감지 (선택)
주의: collision_mask에서 플레이어 자신의 레이어는 반드시 제외하세요. 그렇지 않으면 카메라가 계속 플레이어에 충돌합니다.

TPS 이동 (카메라 기준 방향)

핵심 카메라 방향을 기준으로 수평 이동

@onready var camera_pivot: Node3D = $CameraPivot

@export var speed: float = 5.0

func _physics_process(delta: float):
    # 중력
    if not is_on_floor():
        velocity.y -= gravity * delta

    # 점프
    if Input.is_action_just_pressed("jump") and is_on_floor():
        velocity.y = jump_velocity

    # 입력 벡터
    var input_dir := Input.get_vector("move_left", "move_right", "move_forward", "move_back")

    # 카메라 피봇의 수평 방향 벡터 구하기
    var cam_basis := camera_pivot.global_transform.basis
    var forward := -cam_basis.z  # 카메라 전방
    var right := cam_basis.x      # 카메라 우측

    # Y 성분 제거 후 정규화 (수평 이동만)
    forward.y = 0.0
    right.y = 0.0
    forward = forward.normalized()
    right = right.normalized()

    # 최종 이동 방향
    var direction := (forward * -input_dir.y + right * input_dir.x).normalized()

    if direction:
        velocity.x = direction.x * speed
        velocity.z = direction.z * speed
    else:
        velocity.x = move_toward(velocity.x, 0.0, speed)
        velocity.z = move_toward(velocity.z, 0.0, speed)

    move_and_slide()
Tip: input_dir.y에 마이너스를 곱하는 이유는 Input.get_vector에서 forward가 음수 Y로 반환되기 때문입니다. 카메라의 -Z(전방)와 맞추기 위한 부호 보정입니다.

@onready var model: Node3D = $PlayerModel

func update_rotation(
    direction: Vector3, delta: float
):
    if direction.length() > 0.1:

        var target_angle = atan2(
            direction.x, direction.z
        )

            model.rotation.y,
            target_angle,

# 완성된 TPS 회전 코드 (physics_process 내부)
func _physics_process(delta: float):
    # ... 이동 코드로 direction 계산 ...

    if direction:
        velocity.x = direction.x * speed
        velocity.z = direction.z * speed

계단 오르기 (Step Up)

고급 작은 턱과 계단을 부드럽게 오르기

@export var max_step_height: float = 0.3  # 최대 오를 수 있는 턱 높이

func step_up_stairs(delta: float):
    if velocity.length() < 0.1 or not is_on_floor():
        return

    var motion = velocity * delta
    var step_up = Vector3.UP * max_step_height

    # 1단계: 위로 이동 테스트
    var up_result = PhysicsTestMotionResult3D.new()
    var up_params = PhysicsTestMotionParameters3D.new()
    up_params.from = global_transform
    up_params.motion = step_up

    if PhysicsServer3D.body_test_motion(get_rid(), up_params, up_result):
        return  # 위에 공간이 없으면 중단

    # 2단계: 위에서 전진 테스트
    var raised_transform = global_transform
    raised_transform.origin += step_up
    var fwd_params = PhysicsTestMotionParameters3D.new()
    fwd_params.from = raised_transform
    fwd_params.motion = motion

    if not PhysicsServer3D.body_test_motion(get_rid(), fwd_params):
        # 3단계: 아래로 스냅
        var fwd_transform = raised_transform
        fwd_transform.origin += motion
        var down_params = PhysicsTestMotionParameters3D.new()
        down_params.from = fwd_transform
        down_params.motion = -step_up
        var down_result = PhysicsTestMotionResult3D.new()

        if PhysicsServer3D.body_test_motion(get_rid(), down_params, down_result):
            global_transform.origin = fwd_transform.origin + down_result.get_travel()
Tip: max_step_height를 0.25~0.4 사이로 설정하면 일반적인 게임 계단에 적합합니다. 너무 크면 높은 턱을 비정상적으로 오르게 됩니다.

경사면 처리

floor_max_angle과 슬라이딩으로 자연스러운 경사 이동

extends CharacterBody3D

func _ready():
    # 경사면 관련 CharacterBody3D 설정
    floor_max_angle = deg_to_rad(45)   # 최대 오를 수 있는 경사 (45도)
    floor_snap_length = 0.5             # 경사면에 붙어있는 힘
    floor_stop_on_slope = true          # 경사면에서 멈출 때 미끄러지지 않음
    floor_block_on_wall = true          # 벽에 부딪히면 정지

@export var slide_speed: float = 3.0

func _physics_process(delta: float):
    # ... 기본 이동 코드 ...

    # 급경사 미끄러짐 구현
    if is_on_floor():
        var floor_normal = get_floor_normal()
        var floor_angle = floor_normal.angle_to(Vector3.UP)

        # 최대 경사각 초과 시 미끄러짐
        if floor_angle > floor_max_angle:
            # 경사면 아래 방향 계산
            var slide_dir = Vector3(floor_normal.x, 0, floor_normal.z).normalized()
            velocity += slide_dir * slide_speed * delta

floor_snap_length 0.1 바닥에 달라붙는 스냅 거리

벽 슬라이딩

기능 벽에 붙어 미끄러지기 + 벽 점프

@export var wall_slide_gravity: float = 2.0    # 벽 슬라이드 시 감소된 중력
@export var wall_jump_force: float = 6.0        # 벽 점프 수직 속도
@export var wall_jump_horizontal: float = 4.0   # 벽 점프 수평 속도

var is_wall_sliding: bool = false

func _physics_process(delta: float):
    # 벽 슬라이딩 판정
    is_wall_sliding = (is_on_wall_only()
        and not is_on_floor()
        and velocity.y < 0.0)

    # 중력 적용 (벽 슬라이딩 시 감소)
    if not is_on_floor():
        if is_wall_sliding:
            velocity.y = max(velocity.y, -wall_slide_gravity)
        else:
            velocity.y -= gravity * delta

    # 벽 점프
    if Input.is_action_just_pressed("jump"):
        if is_on_floor():
            velocity.y = jump_velocity
        elif is_wall_sliding:
            # 벽 반대 방향으로 튕겨나감
            var wall_normal = get_wall_normal()
            velocity.y = wall_jump_force
            velocity.x = wall_normal.x * wall_jump_horizontal
            velocity.z = wall_normal.z * wall_jump_horizontal
            is_wall_sliding = false

    # ... 이동 코드 ...
    move_and_slide()

벽 관련 함수

  • is_on_wall() - 벽에 접촉 중 (바닥도 가능)
  • is_on_wall_only() - 벽에만 접촉 (바닥 아님)
  • get_wall_normal() - 접촉한 벽의 법선 벡터
  • get_slide_collision() - 충돌 정보 상세 확인

완성된 FPS 컨트롤러 전체 코드

실전 모든 기능을 통합한 최종 스크립트

extends CharacterBody3D

# --- 노드 참조 ---
@onready var camera_pivot: Node3D = $CameraPivot
@onready var camera: Camera3D = $CameraPivot/Camera3D
@onready var collision: CollisionShape3D = $CollisionShape3D
@onready var head_check: RayCast3D = $HeadCheck

# --- 이동 ---
@export_group("Movement")
@export var walk_speed: float = 5.0
@export var sprint_speed: float = 8.5
@export var crouch_speed: float = 2.5
@export var acceleration: float = 10.0
@export var deceleration: float = 8.0
@export var air_control: float = 0.3

# --- 점프 ---
@export_group("Jump")
@export var jump_velocity: float = 4.5
@export var coyote_time: float = 0.15

# --- 카메라 ---
@export_group("Camera")
@export var mouse_sensitivity: float = 0.002
@export var normal_fov: float = 75.0
@export var sprint_fov: float = 85.0

# --- 상태 ---
var gravity: float = ProjectSettings.get_setting("physics/3d/default_gravity")
var coyote_timer: float = 0.0
var is_sprinting: bool = false
var is_crouching: bool = false

func _ready():
    Input.set_mouse_mode(Input.MOUSE_MODE_CAPTURED)

func _unhandled_input(event: InputEvent):
    # 마우스 룩
    if event is InputEventMouseMotion:
        if Input.get_mouse_mode() == Input.MOUSE_MODE_CAPTURED:
            rotate_y(-event.relative.x * mouse_sensitivity)
            camera_pivot.rotate_x(-event.relative.y * mouse_sensitivity)
            camera_pivot.rotation.x = clamp(
                camera_pivot.rotation.x, deg_to_rad(-89), deg_to_rad(89))
    # ESC / 클릭 토글
    if event.is_action_pressed("ui_cancel"):
        Input.set_mouse_mode(Input.MOUSE_MODE_VISIBLE)
    elif event is InputEventMouseButton and event.pressed:
        if Input.get_mouse_mode() == Input.MOUSE_MODE_VISIBLE:
            Input.set_mouse_mode(Input.MOUSE_MODE_CAPTURED)

func _physics_process(delta: float):
    # 중력 + 코요테 타임
    if not is_on_floor():
        velocity.y -= gravity * delta
        coyote_timer -= delta
    else:
        coyote_timer = coyote_time
    # 점프
    if Input.is_action_just_pressed("jump"):
        if is_on_floor() or coyote_timer > 0.0:
            velocity.y = jump_velocity
            coyote_timer = 0.0
    # 스프린트 / 웅크리기 판정
    is_sprinting = Input.is_action_pressed("sprint") and not is_crouching
    handle_crouch(delta)
    # 속도 결정
    var current_speed = sprint_speed if is_sprinting else (
        crouch_speed if is_crouching else walk_speed)
    # 방향 계산
    var input_dir := Input.get_vector(
        "move_left", "move_right", "move_forward", "move_back")
    var direction := (transform.basis * Vector3(
        input_dir.x, 0, input_dir.y)).normalized()
    # 가속/감속
    var accel = acceleration if is_on_floor() else acceleration * air_control
    if direction:
        velocity.x = lerp(velocity.x, direction.x * current_speed, accel * delta)
        velocity.z = lerp(velocity.z, direction.z * current_speed, accel * delta)
    else:
        velocity.x = lerp(velocity.x, 0.0, deceleration * delta)
        velocity.z = lerp(velocity.z, 0.0, deceleration * delta)
    # FOV
    var target_fov = sprint_fov if is_sprinting else normal_fov
    camera.fov = lerp(camera.fov, target_fov, 5.0 * delta)
    move_and_slide()

func handle_crouch(delta: float):
    var wants_crouch = Input.is_action_pressed("crouch")
    if wants_crouch and not is_crouching:
        is_crouching = true
        (collision.shape as CapsuleShape3D).height = 1.0
        collision.position.y = 0.5
    elif not wants_crouch and is_crouching and not head_check.is_colliding():
        is_crouching = false
        (collision.shape as CapsuleShape3D).height = 1.8
        collision.position.y = 0.9
    var target_y = 0.8 if is_crouching else 1.6
    camera_pivot.position.y = lerp(camera_pivot.position.y, target_y, 10.0 * delta)
