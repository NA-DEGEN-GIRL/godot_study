AnimationPlayer 개요

AnimationPlayer = 범용 애니메이션 엔진

씬 트리 내 모든 노드의 모든 속성을 시간 축에 따라 키프레임으로 제어할 수 있는 강력한 애니메이션 도구입니다. 하나의 AnimationPlayer로 여러 노드를 동시에 애니메이션합니다.

주요 특징

  • 모든 노드의 모든 속성을 애니메이션 가능
  • 여러 트랙을 동시에 재생
  • 트랙 종류: 속성, 메서드, 오디오, Bezier
  • 에디터에서 시각적으로 편집
  • 코드에서 재생/정지/큐잉 제어
# AnimationPlayer 기본 사용
@onready var anim = $AnimationPlayer

func _ready():
    # 애니메이션 재생
    anim.play("idle")

    # 시그널 연결
    anim.animation_finished.connect(
        _on_animation_finished
    )

func _on_animation_finished(name):
    print(name, " 완료!")
    if name == "attack":
        anim.play("idle")
Tip: 에디터 하단의 Animation 패널에서 새 애니메이션을 만들고, 인스펙터에서 속성 옆의 열쇠 아이콘을 클릭하면 현재 값이 키프레임으로 기록됩니다.

키프레임 애니메이션

# 코드로 Animation 리소스 생성
func create_bounce_animation() -> Animation:
    var anim = Animation.new()
    anim.length = 1.0
    anim.loop_mode = Animation.LOOP_LINEAR

    # 위치 Y 트랙 추가
    var track_idx = anim.add_track(Animation.TYPE_VALUE)
    anim.track_set_path(track_idx, "Sprite2D:position:y")

    # 키프레임 삽입
    anim.track_insert_key(track_idx, 0.0, 0.0)     # 시작: y=0
    anim.track_insert_key(track_idx, 0.5, -30.0)    # 중간: y=-30 (위로)
    anim.track_insert_key(track_idx, 1.0, 0.0)      # 끝: y=0

    # 보간 모드 설정
    anim.track_set_interpolation_type(
        track_idx, Animation.INTERPOLATION_CUBIC
    )

    return anim

func _ready():
    var anim_player = $AnimationPlayer
    var bounce = create_bounce_animation()

    # 애니메이션 라이브러리에 추가
    var lib = anim_player.get_animation_library("")
    lib.add_animation("bounce", bounce)
    anim_player.play("bounce")
보간 타입설명사용 예
INTERPOLATION_NEAREST즉시 전환 (스텝)스프라이트 전환, ON/OFF
INTERPOLATION_LINEAR선형 보간일정한 속도 이동
INTERPOLATION_CUBIC큐빅 보간 (부드럽게)자연스러운 움직임

AnimatedSprite2D

설정 방법

  1. AnimatedSprite2D 노드 추가
  2. 인스펙터에서 SpriteFrames 새로 생성
  3. SpriteFrames 에디터에서 애니메이션 추가
  4. 각 애니메이션에 프레임 이미지 드래그
  5. FPS, 루프 설정 조정

Tip: 스프라이트 시트를 사용하면 하나의 이미지에서 여러 프레임을 자동으로 잘라낼 수 있습니다. SpriteFrames 에디터에서 격자 아이콘을 클릭하세요.

SpriteFrames 리소스

# 코드로 SpriteFrames 생성
func create_sprite_frames() -> SpriteFrames:
    var frames = SpriteFrames.new()

    # 기본 "default" 애니메이션 제거
    frames.remove_animation("default")

    # idle 애니메이션 (4프레임, 루프)
    frames.add_animation("idle")
    frames.set_animation_speed("idle", 8.0)  # 8 FPS
    frames.set_animation_loop("idle", true)
    for i in range(4):
        var tex = load("res://sprites/player/idle_%d.png" % i)
        frames.add_frame("idle", tex)

    # walk 애니메이션 (6프레임, 루프)
    frames.add_animation("walk")
    frames.set_animation_speed("walk", 12.0)  # 12 FPS
    frames.set_animation_loop("walk", true)
    for i in range(6):
        var tex = load("res://sprites/player/walk_%d.png" % i)
        frames.add_frame("walk", tex)

    # attack 애니메이션 (5프레임, 루프 안함)
    frames.add_animation("attack")
    frames.set_animation_speed("attack", 15.0)
    frames.set_animation_loop("attack", false)
    for i in range(5):
        var tex = load("res://sprites/player/attack_%d.png" % i)
        frames.add_frame("attack", tex)

    return frames

AnimatedSprite2D vs AnimationPlayer + Sprite2D

AnimatedSprite2D - 프레임 교체만 필요할 때 간편. AnimationPlayer - 위치, 크기, 색상 등 여러 속성을 동시에 애니메이션할 때 적합. 복잡한 캐릭터 애니메이션은 AnimationPlayer가 유리합니다.

애니메이션 재생 제어

# AnimationPlayer 재생 제어 함수들
@onready var anim = $AnimationPlayer

func _ready():
    anim.play("idle")

# 기본 재생 제어
func play_animation(name: String):
    anim.play(name)              # 즉시 재생
    # anim.play(name, -1, 2.0)   # 2배속 재생
    # anim.play(name, -1, -1.0)  # 역재생

func stop_animation():
    anim.stop()                  # 정지 (처음으로)

func pause_animation():
    anim.pause()                 # 일시정지 (현재 위치 유지)

func resume_animation():
    anim.play()                  # 마지막 위치에서 재개

# 큐잉: 현재 애니메이션 끝나면 자동 재생
func attack_combo():
    anim.play("attack_1")
    anim.queue("attack_2")      # attack_1 끝나면 attack_2
    anim.queue("idle")           # attack_2 끝나면 idle

# 특정 시간으로 이동
func seek_to(time: float):
    anim.seek(time, true)        # true = 즉시 업데이트

# 재생 속도 조절
func set_speed(scale: float):
    anim.speed_scale = scale     # 0.5 = 반속, 2.0 = 2배속
함수동작활용 예
play(name)즉시 재생 (현재 중단)상태 전환
queue(name)현재 끝난 후 재생연속 콤보
stop()정지, 처음으로 초기화애니메이션 리셋
pause() / play()일시정지 / 재개컷씬 제어
seek(time)특정 시간으로 점프미리보기, 스킵

Tween 클래스

Tween = 코드 기반 간편 애니메이션

에디터 없이 코드 몇 줄로 속성 보간 애니메이션을 만들 수 있습니다. UI 전환, 이펙트, 임시 연출에 매우 유용합니다. Godot 4에서 완전히 새롭게 설계되었습니다.

    tw.tween_property(self, "position:y", -100, 0.5)
    tw.tween_callback(func(): print("최고점 도달!"))
    tw.tween_property(self, "position:y", 0, 0.5)

# 3. tween_method: 커스텀 함수에 보간값 전달
func example_method():
    var tw = create_tween()
    # (함수, 시작값, 끝값, 시간)
    tw.tween_method(set_hp_display, 100.0, 0.0, 2.0)

func set_hp_display(value: float):

    $HPBar.value = value

# 4. tween_interval: 대기
func example_interval():
    var tw = create_tween()
    tw.tween_property(self, "modulate", Color.RED, 0.1)
    tw.tween_interval(0.5)  # 0.5초 대기
    tw.tween_property(self, "modulate", Color.WHITE, 0.1)

이징 함수 (Easing)

이징 = 애니메이션의 느낌을 결정하는 곡선

set_trans()로 곡선 형태를, set_ease()로 가감속 방향을 설정합니다.

타입효과

func bounce_in():

    tw.tween_property(

     .set_ease(Tween.EASE_OUT)

애니메이션 상태 머신 개념

# 코드로 구현하는 애니메이션 상태 머신
extends CharacterBody2D

enum State { IDLE, WALK, JUMP, FALL, ATTACK, HURT }
var current_state: State = State.IDLE

@onready var anim = $AnimationPlayer
@onready var sprite = $Sprite2D

func _physics_process(delta):
    var new_state = determine_state()
    if new_state != current_state:
        exit_state(current_state)
        current_state = new_state
        enter_state(current_state)

func determine_state() -> State:
    if current_state == State.HURT:
        if anim.is_playing():
            return State.HURT
    if current_state == State.ATTACK:
        if anim.is_playing():
            return State.ATTACK
    if not is_on_floor():
        return State.FALL if velocity.y > 0 else State.JUMP
    if abs(velocity.x) > 10:
        return State.WALK
    return State.IDLE

func enter_state(state: State):
    match state:
        State.IDLE:   anim.play("idle")
        State.WALK:   anim.play("walk")
        State.JUMP:   anim.play("jump")
        State.FALL:   anim.play("fall")
        State.ATTACK: anim.play("attack")
        State.HURT:   anim.play("hurt")

func exit_state(state: State):
    pass  # 필요시 상태 종료 로직

AnimationTree 소개

AnimationTree = 시각적 애니메이션 제어 시스템

AnimationPlayer의 애니메이션들을 상태 머신, 블렌드 트리 등의 노드 그래프로 연결하여 복잡한 전환과 블렌딩을 관리합니다.

구성 요소

  • AnimationNodeStateMachine 상태 기반 전환
  • AnimationNodeBlendTree 블렌드 트리
  • AnimationNodeBlendSpace2D 2D 블렌드 스페이스
  • AnimationNodeTransition 전환 노드
# AnimationTree 기본 설정
@onready var anim_tree = $AnimationTree
@onready var state_machine: AnimationNodeStateMachinePlayback

func _ready():
    anim_tree.active = true
    state_machine = anim_tree.get(
        "parameters/playback"
    )

# 상태 전환
func set_animation_state(state: String):
    state_machine.travel(state)

# 파라미터 제어
func _physics_process(delta):
    anim_tree.set(
        "parameters/walk_blend/blend_position",
        velocity.length() / max_speed
    )
Tip: AnimationTree는 AnimationPlayer와 함께 사용해야 합니다. AnimationPlayer에 애니메이션을 먼저 만들고, AnimationTree에서 재생 로직을 관리하는 구조입니다.

# AnimationTree 구조:

#   │   └── (0, 1)   idle_down
#   └── walk (BlendSpace2D)

#       ├── (1, 0)   walk_right
#       ├── (0, -1)  walk_up
#       └── (0, 1)   walk_down

코드에서 블렌드 제어

# 블렌드 스페이스 파라미터 업데이트
@onready var anim_tree = $AnimationTree
var blend_path_idle = \
    "parameters/idle/blend_position"
var blend_path_walk = \
    "parameters/walk/blend_position"

func _physics_process(delta):
    var input = Input.get_vector(
        "left", "right", "up", "down"
    )

    if input != Vector2.ZERO:
        # 이동 중: walk 상태
        anim_tree.set(
            blend_path_walk, input
        )
        state_machine.travel("walk")
    else:
        # 정지: idle 상태
        state_machine.travel("idle")

    # idle도 마지막 방향 유지
    if input != Vector2.ZERO:
        anim_tree.set(
            blend_path_idle, input
        )

실전: 캐릭터 애니메이션 시스템

# player.gd - 완전한 캐릭터 애니메이션 시스템
extends CharacterBody2D

const SPEED = 200.0
const JUMP_VELOCITY = -350.0

@onready var anim_player = $AnimationPlayer
@onready var sprite = $Sprite2D

var is_attacking := false
var facing_right := true

func _ready():
    anim_player.animation_finished.connect(_on_anim_finished)

func _physics_process(delta):
    # 중력
    if not is_on_floor():
        velocity.y += 980 * delta

    # 입력 처리
    if not is_attacking:
        handle_input()

    move_and_slide()
    update_animation()

func handle_input():
    var direction = Input.get_axis("left", "right")
    velocity.x = direction * SPEED

    # 방향 전환
    if direction > 0:
        sprite.flip_h = false
        facing_right = true
    elif direction < 0:
        sprite.flip_h = true
        facing_right = false

    # 점프
    if Input.is_action_just_pressed("jump") and is_on_floor():
        velocity.y = JUMP_VELOCITY

    # 공격
    if Input.is_action_just_pressed("attack"):
        is_attacking = true
        velocity.x = 0

func update_animation():
    if is_attacking:
        anim_player.play("attack")
    elif not is_on_floor():
        anim_player.play("jump" if velocity.y < 0 else "fall")
    elif abs(velocity.x) > 10:
        anim_player.play("walk")
    else:
        anim_player.play("idle")

func _on_anim_finished(name: StringName):
    if name == "attack":
        is_attacking = false

# 페이드 전환
func fade_to_scene(scene_path: String):
    color_rect.visible = true
    # 페이드 아웃 (어두워짐)
    var tw = create_tween()
    tw.tween_property(color_rect, "color:a", 1.0, 0.5).from(0.0)
    tw.tween_callback(func():
        transition_midpoint.emit()
        get_tree().change_scene_to_file(scene_path)
    )
    tw.tween_interval(0.1)  # 씬 로드 대기
    # 페이드 인 (밝아짐)
    tw.tween_property(color_rect, "color:a", 0.0, 0.5)
    tw.tween_callback(func(): color_rect.visible = false)

# 커스텀 색상 전환
func flash_white(duration: float = 0.3):
    color_rect.visible = true
    color_rect.color = Color.WHITE
    var tw = create_tween()
    tw.tween_property(color_rect, "color:a", 1.0, duration * 0.3).from(0.0)
    tw.tween_property(color_rect, "color:a", 0.0, duration * 0.7)
    tw.tween_callback(func(): color_rect.visible = false)
