GridMap 개요

3D 타일맵 격자 기반 3D 레벨 구축 시스템

GridMap의 구조

  • GridMap 노드 - 3D 격자를 관리하는 컨테이너
  • MeshLibrary - 배치 가능한 메시 모음 리소스
  • Cell Size - 격자 한 칸의 크기 (Vector3)
  • Cell Item - 각 격자에 놓인 메시의 인덱스
비유: MeshLibrary는 레고 블록 세트이고, GridMap은 레고를 올려놓는 베이스 판입니다. 격자 위에 블록을 하나씩 배치하여 맵을 만듭니다.

주요 프로퍼티

# GridMap 인스펙터 주요 설정
# MeshLibrary: 사용할 메시 라이브러리
# Cell > Size: Vector3(2, 2, 2) - 셀 크기
# Cell > Octant Size: 8 - 렌더링 청크 크기
# Cell > Center X/Y/Z: 셀 중심 정렬

# 씬 트리 구성 예시
# Level (Node3D)
#   +-- GridMap         (바닥/벽 타일)
#   +-- GridMap2        (장식용 오브젝트)
#   +-- DirectionalLight3D
#   +-- Camera3D
주의: Cell Size는 MeshLibrary에 포함된 메시의 실제 크기와 일치해야 이음새 없이 배치됩니다.

# MeshLibrary를 코드로 생성
func create_mesh_library() -> MeshLibrary:
    var lib = MeshLibrary.new()

    # 바닥 타일 추가 (인덱스 0)
    var floor_mesh = BoxMesh.new()
    floor_mesh.size = Vector3(2, 0.2, 2)
    lib.create_item(0)
    lib.set_item_name(0, "Floor")
    lib.set_item_mesh(0, floor_mesh)

    # 충돌 형태 추가
    var shape = BoxShape3D.new()
    shape.size = Vector3(2, 0.2, 2)
    lib.set_item_shapes(0, [
        Transform3D.IDENTITY, shape
    ])

    # 벽 타일 추가 (인덱스 1)
    var wall_mesh = BoxMesh.new()
    wall_mesh.size = Vector3(2, 2, 0.2)
    lib.create_item(1)
    lib.set_item_name(1, "Wall")
    lib.set_item_mesh(1, wall_mesh)

    return lib

코드로 GridMap 조작

GDScript 동적 레벨 생성과 수정

extends Node3D

@onready var grid_map: GridMap = $GridMap

# 메시 인덱스 상수
const FLOOR := 0
const WALL := 1
const PILLAR := 2

func _ready():
    # 10x10 바닥 생성
    for x in range(-5, 5):
        for z in range(-5, 5):
            grid_map.set_cell_item(Vector3i(x, 0, z), FLOOR)

    # 외벽 생성
    for x in range(-5, 5):
        grid_map.set_cell_item(Vector3i(x, 1, -5), WALL)
        grid_map.set_cell_item(Vector3i(x, 1, 4), WALL)
    for z in range(-5, 5):
        grid_map.set_cell_item(Vector3i(-5, 1, z), WALL)
        grid_map.set_cell_item(Vector3i(4, 1, z), WALL)

# 셀 삭제
func remove_cell(pos: Vector3i):
    grid_map.set_cell_item(pos, GridMap.INVALID_CELL_ITEM)

# 월드 좌표를 격자 좌표로 변환
func world_to_grid(world_pos: Vector3) -> Vector3i:
    return grid_map.local_to_map(grid_map.to_local(world_pos))

# 격자 좌표를 월드 좌표로 변환
func grid_to_world(grid_pos: Vector3i) -> Vector3:
    return grid_map.to_global(grid_map.map_to_local(grid_pos))

# 배치된 모든 셀 순회
func count_cells():
    var cells = grid_map.get_used_cells()
    print("총 배치된 셀: ", cells.size())
    for cell in cells:
        var item_id = grid_map.get_cell_item(cell)
        var orientation = grid_map.get_cell_item_orientation(cell)
        print("  위치: %v, 아이템: %d, 방향: %d" % [cell, item_id, orientation])
활용: 절차적 던전 생성 알고리즘(BSP, 드렁큰 워크 등)과 결합하면 매번 다른 레벨을 자동 생성할 수 있습니다.

# 코드로 CSG 불리언 연산
func create_wall_with_door():
    var combiner = CSGCombiner3D.new()
    combiner.use_collision = true
    add_child(combiner)

    # 벽 본체
    var wall = CSGBox3D.new()
    wall.size = Vector3(6, 3, 0.3)
    combiner.add_child(wall)

    # 문 구멍 (Subtraction)
    var door = CSGBox3D.new()
    door.size = Vector3(1.2, 2.2, 0.5)
    door.position = Vector3(0, -0.4, 0)
    door.operation = CSGShape3D.OPERATION_SUBTRACTION
    combiner.add_child(door)

핵심 원리

NavigationRegion3D는 자식 노드의 메시 형태를 분석하여 걸어다닐 수 있는 표면(NavMesh)을 계산합니다. 이 표면 위에서만 경로 탐색이 가능합니다.

코드로 설정

# NavigationRegion3D 코드 설정
@onready var nav_region = $NavigationRegion3D

func _ready():
    # NavigationMesh 생성 및 할당
    var nav_mesh = NavigationMesh.new()

    # 에이전트 파라미터 설정
    nav_mesh.agent_radius = 0.5
    nav_mesh.agent_height = 1.8
    nav_mesh.agent_max_climb = 0.3
    nav_mesh.agent_max_slope = 45.0

    # 셀 크기 (정밀도)
    nav_mesh.cell_size = 0.25
    nav_mesh.cell_height = 0.25

    nav_region.navigation_mesh = nav_mesh

    # 베이킹 실행
    nav_region.bake_navigation_mesh()

# 베이킹 완료 시그널
func _on_nav_region_bake_finished():
    print("NavMesh 베이킹 완료!")

NavigationAgent3D

AI 이동 개별 캐릭터의 경로 탐색 에이전트

# enemy.gd - NavigationAgent3D 기본 사용법
extends CharacterBody3D

@export var speed: float = 4.0
@onready var nav_agent: NavigationAgent3D = $NavigationAgent3D

func _ready():
    # 에이전트 설정
    nav_agent.path_desired_distance = 0.5    # 경유점 도달 판정 거리
    nav_agent.target_desired_distance = 0.5  # 최종 목표 도달 판정 거리
    nav_agent.max_speed = speed

    # 경로 재계산 간격 (성능 최적화)
    nav_agent.path_postprocessing = NavigationPathQueryParameters3D.PATH_POSTPROCESSING_EDGECENTERED

func set_target(target_pos: Vector3):
    nav_agent.target_position = target_pos

func _physics_process(delta):
    if nav_agent.is_navigation_finished():
        return

    # 다음 경로 위치 가져오기
    var next_pos = nav_agent.get_next_path_position()

    # 방향 계산 및 이동
    var direction = (next_pos - global_position).normalized()
    velocity = direction * speed

    # 이동 방향으로 캐릭터 회전
    if direction.length() > 0.1:
        var look_target = global_position + direction
        look_at(look_target, Vector3.UP)

    move_and_slide()
Tip: NavigationAgent3D의 path_desired_distance를 너무 작게 설정하면 에이전트가 경유점에서 방황할 수 있습니다. 캐릭터 반경의 1~2배 정도가 적당합니다.

경로 탐색 실전 패턴

추적, 순찰, 도달 감지 구현

플레이어 추적

# enemy_chase.gd
extends CharacterBody3D

@export var speed: float = 3.5
@export var chase_range: float = 15.0
@onready var nav: NavigationAgent3D = $NavigationAgent3D
var player: Node3D

func _ready():
    player = get_tree().get_first_node_in_group("player")
    # 0.3초마다 경로 갱신 (매 프레임 X)
    var timer = Timer.new()
    timer.wait_time = 0.3
    timer.timeout.connect(_update_target)
    add_child(timer)
    timer.start()

    # 목표 도달 시그널
    nav.navigation_finished.connect(
        _on_navigation_finished
    )

func _update_target():
    if not player:
        return
    var dist = global_position.distance_to(
        player.global_position
    )
    if dist < chase_range:
        nav.target_position = player.global_position

func _physics_process(delta):
    if nav.is_navigation_finished():
        return
    var next = nav.get_next_path_position()
    var dir = (next - global_position).normalized()
    velocity = dir * speed
    move_and_slide()

func _on_navigation_finished():
    print("목표 도달!")

순찰 시스템

# enemy_patrol.gd
extends CharacterBody3D

@export var patrol_points: Array[Marker3D] = []
@export var speed: float = 2.0
@export var wait_time: float = 2.0

@onready var nav: NavigationAgent3D = $NavigationAgent3D
var current_index: int = 0
var waiting: bool = false

func _ready():
    if patrol_points.size() > 0:
        _go_to_next_point()
    nav.navigation_finished.connect(
        _on_point_reached
    )

func _go_to_next_point():
    var target = patrol_points[current_index]
    nav.target_position = target.global_position

func _on_point_reached():
    waiting = true
    await get_tree().create_timer(wait_time).timeout
    current_index = (current_index + 1) % patrol_points.size()
    _go_to_next_point()
    waiting = false

func _physics_process(delta):
    if nav.is_navigation_finished() or waiting:
        return
    var next = nav.get_next_path_position()
    var dir = (next - global_position).normalized()
    velocity = dir * speed
    if dir.length() > 0.1:
        look_at(global_position + dir, Vector3.UP)
    move_and_slide()

에이전트 회피

RVO 알고리즘 여러 에이전트가 서로 피하며 이동

# avoidance_enemy.gd - 회피 기능 활성화
extends CharacterBody3D

@export var speed: float = 3.0
@onready var nav: NavigationAgent3D = $NavigationAgent3D

func _ready():
    # 회피 설정
    nav.avoidance_enabled = true
    nav.radius = 0.5                  # 에이전트 회피 반경
    nav.max_speed = speed             # 최대 속도 (회피 계산에 사용)
    nav.neighbor_distance = 10.0      # 이웃 감지 거리
    nav.max_neighbors = 10            # 고려할 최대 이웃 수
    nav.time_horizon_agents = 1.0     # 에이전트 회피 예측 시간
    nav.time_horizon_obstacles = 0.5  # 장애물 회피 예측 시간

    # 회피 결과를 시그널로 받음
    nav.velocity_computed.connect(_on_velocity_computed)

func _physics_process(delta):
    if nav.is_navigation_finished():
        return

    var next_pos = nav.get_next_path_position()
    var direction = (next_pos - global_position).normalized()

    # 원하는 속도를 에이전트에 전달 (회피 계산 요청)
    nav.velocity = direction * speed

# 회피가 적용된 안전 속도를 받아서 실제 이동에 사용
func _on_velocity_computed(safe_velocity: Vector3):
    velocity = safe_velocity
    move_and_slide()
주의: avoidance_enabled가 true이면 반드시 velocity_computed 시그널을 사용하세요. 직접 velocity를 설정하면 회피가 무시됩니다. 또한 에이전트 수가 많으면 neighbor_distance와 max_neighbors를 줄여 성능을 확보하세요.

내비게이션 레이어

레이어 설계 예시

레이어 비트 용도

NavigationRegion3D의 navigation_layers는 해당 영역이 속하는 레이어이고, NavigationAgent3D의 navigation_layers는 탐색할 레이어 마스크입니다.

    nav_region_air.navigation_layers = 2

# 보행 유닛 에이전트 설정
func setup_walking_agent(agent: NavigationAgent3D):
    agent.navigation_layers = 1  # 바닥만 사용

# 비행 유닛 에이전트 설정
func setup_flying_agent(agent: NavigationAgent3D):
    agent.navigation_layers = 3  # 바닥 + 공중 (1|2)

# 런타임에 레이어 변경 (예: 비행 모드 전환)
func toggle_flying(agent: NavigationAgent3D, can_fly: bool):
    if can_fly:
        agent.set_navigation_layer_value(2, true)
    else:
        agent.set_navigation_layer_value(2, false)

# 바리케이드 - NavMesh에 영향
extends StaticBody3D

@onready var obstacle = $NavigationObstacle3D

func _ready():
    # affect_navigation_mesh를 true로 설정하면
    # NavMesh 자체를 업데이트합니다
    obstacle.affect_navigation_mesh = true
    obstacle.carve_navigation_mesh = true

    # 장애물 정점 정의 (다각형)
    obstacle.vertices = PackedVector3Array([
        Vector3(-1, 0, -1),
        Vector3(1, 0, -1),
        Vector3(1, 0, 1),
        Vector3(-1, 0, 1),
    ])

# 동적으로 장애물 추가
func spawn_obstacle(pos: Vector3):
    var obs = NavigationObstacle3D.new()
    obs.radius = 2.0
    obs.avoidance_enabled = true
    obs.position = pos
    add_child(obs)
carve vs avoidance: carve_navigation_mesh는 NavMesh 폴리곤을 잘라내어 경로 자체를 변경합니다. avoidance는 경로는 유지하되 에이전트가 실시간으로 피합니다.

Marker3D 스폰 포인트

레벨 디자인 위치 마커로 스폰/웨이포인트 설정

스폰 포인트 시스템

# spawn_manager.gd
extends Node3D

@export var enemy_scene: PackedScene
@export var max_enemies: int = 10
@export var spawn_interval: float = 3.0

var spawn_points: Array[Marker3D] = []
var active_enemies: int = 0

func _ready():
    # 그룹으로 스폰 포인트 수집
    for marker in get_tree().get_nodes_in_group(
        "enemy_spawn"
    ):
        if marker is Marker3D:
            spawn_points.append(marker)

    # 스폰 타이머 시작
    var timer = Timer.new()
    timer.wait_time = spawn_interval
    timer.timeout.connect(_spawn_enemy)
    add_child(timer)
    timer.start()

func _spawn_enemy():
    if active_enemies >= max_enemies:
        return
    if spawn_points.is_empty():
        return

    # 랜덤 스폰 포인트 선택
    var point = spawn_points.pick_random()
    var enemy = enemy_scene.instantiate()
    enemy.global_transform = point.global_transform
    enemy.tree_exited.connect(
        func(): active_enemies -= 1
    )
    add_child(enemy)
    active_enemies += 1

    data: Dictionary = {}
):

    var tween = create_tween()

        fade, "color:a", 1.0, 0.5
    )
    await tween.finished

    # 씬 전환

    await get_tree().tree_changed

    # 페이드 인
    var tween2 = create_tween()
    tween2.tween_property(
        fade, "color:a", 0.0, 0.5
    )

    return transition_data

종합 예제 모든 도구를 조합한 던전 만들기

# dungeon_level.gd - 던전 레벨 총 구성
extends Node3D

@export var enemy_scene: PackedScene
@export var treasure_scene: PackedScene

@onready var grid_map: GridMap = $GridMap
@onready var nav_region: NavigationRegion3D = $NavigationRegion3D

func _ready():
    _setup_rooms()
    _bake_navigation()
    _spawn_enemies()
    _place_treasures()
    _connect_doors()

func _setup_rooms():
    # GridMap으로 기본 구조 배치 (또는 에디터에서 미리 배치)
    # 복도 연결
    for x in range(-2, 3):
        grid_map.set_cell_item(Vector3i(x, 0, 0), 0)  # 바닥

func _bake_navigation():
    # NavMesh 베이킹 (에디터에서 미리 하는 것이 일반적)
    nav_region.bake_navigation_mesh()
    await nav_region.bake_finished

func _spawn_enemies():
    for spawn in get_tree().get_nodes_in_group("enemy_spawn"):
        var enemy = enemy_scene.instantiate()
        enemy.global_position = spawn.global_position
        $Enemies.add_child(enemy)

func _place_treasures():
    for spawn in get_tree().get_nodes_in_group("item_spawn"):
        var treasure = treasure_scene.instantiate()
        treasure.global_position = spawn.global_position
        $Items.add_child(treasure)

func _connect_doors():
    for door in get_tree().get_nodes_in_group("doors"):
        door.body_entered.connect(
            func(body): _on_door_entered(door, body)
        )

func _on_door_entered(door: Area3D, body: Node3D):
    if body.is_in_group("player"):
        SceneTransition.change_scene(
            door.get_meta("target_scene"),
            {"spawn_point": door.get_meta("target_spawn")}
        )
