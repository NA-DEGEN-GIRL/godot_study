구성 원칙

  • 기능별 분리 - 관련 파일을 한 폴더에
  • 씬+스크립트 동거 - .tscn과 .gd를 나란히
  • 리소스 독립 - 재사용 데이터는 resources/에
  • 에셋 분리 - 원본 에셋은 assets/에
  • autoload 전용 - 싱글톤은 별도 폴더
피해야 할 것: 모든 스크립트를 scripts/ 폴더에 몰아넣기, 깊은 중첩 폴더(5단계 이상), 이름 규칙 없이 파일 생성

씬 구성 원칙

좋은 씬 설계의 3가지 원칙

  1. 단일 책임 - 한 씬은 하나의 역할만 담당
  2. 독립 실행 - 단독으로 F6 실행이 가능해야 함
  3. 시그널 통신 - 부모를 직접 참조하지 않고 시그널로 전달

상태 패턴 (State Machine)

핵심 패턴 캐릭터 행동을 상태별로 분리

# state_machine.gd - 상태 머신 관리자
class_name StateMachine
extends Node

@export var initial_state: State
var current_state: State

func _ready():
    # 모든 자식 State에 state_machine 참조 전달
    for child in get_children():
        if child is State:
            child.state_machine = self
    if initial_state:
        transition_to(initial_state.name)

func _process(delta):
    if current_state:
        current_state.update(delta)

func _physics_process(delta):
    if current_state:
        current_state.physics_update(delta)

func transition_to(state_name: String):
    if current_state:
        current_state.exit()
    current_state = get_node(state_name)
    current_state.enter()

오브젝트 풀링

성능 패턴 자주 생성/파괴되는 객체를 재활용

# object_pool.gd - 범용 오브젝트 풀
class_name ObjectPool
extends Node

@export var scene: PackedScene
@export var pool_size: int = 20

var _pool: Array[Node] = []

func _ready():
    # 미리 생성해두기
    for i in pool_size:
        var obj = scene.instantiate()
        obj.set_process(false)
        obj.set_physics_process(false)
        obj.hide()
        add_child(obj)
        _pool.append(obj)

# 풀에서 꺼내기
func get_object() -> Node:
    for obj in _pool:
        if not obj.visible:
            obj.show()
            obj.set_process(true)
            obj.set_physics_process(true)
            if obj.has_method("activate"):
                obj.call("activate")
            return obj
    # 풀이 부족하면 새로 생성 (자동 확장)
    var obj = scene.instantiate()
    add_child(obj)
    _pool.append(obj)
    return obj

# 풀에 반환하기
func return_object(obj: Node):
    obj.set_process(false)
    obj.set_physics_process(false)
    obj.hide()
    if obj.has_method("deactivate"):
        obj.call("deactivate")
# 사용 예: 총알 발사
@onready var bullet_pool: ObjectPool = $BulletPool

func shoot():
    var bullet = bullet_pool.get_object()
    if bullet:
        bullet.global_position = muzzle.global_position
        bullet.direction = (get_global_mouse_position() - muzzle.global_position).normalized()

컴포넌트 패턴

조합 패턴 기능을 독립 노드로 분리하여 조합

HealthComponent

# health_component.gd
class_name HealthComponent
extends Node

signal health_changed(current, maximum)
signal died

@export var max_hp: int = 100
var current_hp: int

func _ready():
    current_hp = max_hp

func take_damage(amount: int):
    current_hp = max(0, current_hp - amount)
    health_changed.emit(current_hp, max_hp)
    if current_hp <= 0:
        died.emit()

func heal(amount: int):
    current_hp = min(max_hp, current_hp + amount)
    health_changed.emit(current_hp, max_hp)

게임 매니저 패턴

Autoload 게임 전체 흐름과 상태 관리

# autoload/game_manager.gd
extends Node

enum GameState { TITLE, PLAYING, PAUSED, GAME_OVER, LOADING }

signal state_changed(new_state: GameState)

var current_state: GameState = GameState.TITLE
var score: int = 0
var current_level: int = 1

func change_state(new_state: GameState):
    var old_state = current_state
    current_state = new_state
    state_changed.emit(new_state)

    match new_state:
        GameState.TITLE:
            get_tree().paused = false
            get_tree().change_scene_to_file("res://scenes/ui/title_screen.tscn")
        GameState.PLAYING:
            get_tree().paused = false
        GameState.PAUSED:
            get_tree().paused = true
        GameState.GAME_OVER:
            get_tree().paused = true
            SaveManager.save_high_score(score)
        GameState.LOADING:
            pass

func start_game():
    score = 0
    current_level = 1
    change_state(GameState.LOADING)
    await load_level(current_level)
    change_state(GameState.PLAYING)

func next_level():
    current_level += 1
    change_state(GameState.LOADING)
    await load_level(current_level)
    change_state(GameState.PLAYING)

func load_level(level_num: int):
    var path = "res://scenes/levels/level_%d.tscn" % level_num
    get_tree().change_scene_to_file(path)
    await get_tree().tree_changed

func _input(event):
    if event.is_action_pressed("pause") and current_state == GameState.PLAYING:
        change_state(GameState.PAUSED)
    elif event.is_action_pressed("pause") and current_state == GameState.PAUSED:
        change_state(GameState.PLAYING)

내장 프로파일러 사용법

# 코드에서 시간 측정
func heavy_function():
    var start = Time.get_ticks_msec()
    # ... 무거운 작업 ...
    var elapsed = Time.get_ticks_msec() - start
    print("실행 시간: %d ms" % elapsed)

# 마이크로초 단위 정밀 측정
func precise_measurement():
    var start = Time.get_ticks_usec()
    # ... 작업 ...
    var elapsed = Time.get_ticks_usec() - start
    print("실행 시간: %d us (%.2f ms)" % [elapsed, elapsed / 1000.0])
모니터 항목 정상 범위 주의 필요
FPS 60 (안정적) 30 이하
Process Time 16ms 이하 20ms 이상
Draw Calls 100 이하 500 이상
Objects 프로젝트별 상이 급격한 증가 (메모리 누수)

Draw Call 최적화

최적화 기법

  • 스프라이트 아틀라스 - 여러 이미지를 하나의 큰 텍스처로 합침
  • CanvasGroup - 자식 노드를 하나의 드로우 콜로 합침
  • 같은 Z-index로 동일 레이어 배치
  • VisibleOnScreenNotifier2D - 화면 밖 노드 비활성화
  • TileMap - 타일맵은 자동으로 배칭됨

# 플랫폼별 코드 분기 예시
func get_save_path() -> String:
    match OS.get_name():
        "Windows":
            return OS.get_environment("APPDATA") + "/MyGame/"
        "macOS":
            return OS.get_environment("HOME") + "/Library/Application Support/MyGame/"
        "Linux":
            return OS.get_environment("HOME") + "/.local/share/MyGame/"
        _:
            return "user://"

카테고리 옵션 설명
Resources Export Mode 모든 리소스 / 선택 리소스 / 의존성만
Resources Include Filter 포함할 확장자 (*.json, *.cfg)
Resources Exclude Filter 제외할 패턴 (*.test.*, *.tmp)
Script Encryption Key GDScript 코드 보호 (256bit AES)
Binary Embed PCK PCK를 실행파일에 내장 
# 내보내기 시 환경별 분기
func _ready():
    if OS.has_feature("editor"):
        print("에디터에서 실행 중")
    elif OS.has_feature("debug"):
        print("디버그 빌드 실행 중")
    elif OS.has_feature("release"):
        print("릴리스 빌드 실행 중")

    # 플랫폼 feature 확인
    if OS.has_feature("web"):
        disable_file_save()
    if OS.has_feature("mobile"):
        enable_touch_controls()
    if OS.has_feature("pc"):
        enable_keyboard_controls()
Tip: 커스텀 Feature 태그를 프리셋에 추가할 수 있습니다. OS.has_feature("demo")로 데모 버전과 정식 버전을 분기할 때 유용합니다.

프로젝트 완성 체크리스트

기능 / 품질

  • 모든 기능이 의도대로 작동하는가
  • 세이브/로드가 정상 작동하는가
  • 에러 발생 시 적절히 처리되는가
  • 메모리 누수가 없는가 (queue_free 확인)
  • 입력이 응답성이 좋은가
  • 사운드 볼륨이 적절한가

성능

  • 목표 FPS를 유지하는가 (60fps)
  • 씬 전환 시 끊김이 없는가
  • 메모리 사용량이 적절한가

사용자 경험

  • 일시정지/설정 메뉴가 있는가
  • 조작 방법을 안내하는가
  • 다양한 화면 비율에 대응하는가
  • 폰트 크기가 읽기 편한가

배포

  • 게임 제목과 아이콘이 설정되었는가
  • 디버그 출력(print)을 정리했는가
  • 테스트/임시 코드를 제거했는가
  • 라이선스와 크레딧을 표기했는가
  • 각 플랫폼에서 테스트했는가
