중력 적용 (플랫포머)

extends CharacterBody2D

@export var speed := 300.0
@export var gravity := 980.0        # 픽셀/초^2
# 또는 프로젝트 설정에서 가져오기:
# var gravity = ProjectSettings.get_setting("physics/2d/default_gravity")

func _physics_process(delta):
    # 바닥에 있지 않을 때만 중력 적용
    if not is_on_floor():
        velocity.y += gravity * delta   # 가속도 = 속도 증가
        # 최대 낙하 속도 제한 (선택)
        velocity.y = min(velocity.y, 800.0)

    # 좌우 이동
    var dir = Input.get_axis("move_left", "move_right")
    velocity.x = dir * speed

    move_and_slide()

중력 공식 이해

velocity.y += gravity * delta
delta가 0.016(60fps)이면: 매 프레임 약 15.7 px/s 씩 낙하 속도 증가
중력은 가속도이므로, 시간이 지날수록 점점 빨라짐 (자연스러운 낙하)

점프 구현

extends CharacterBody2D

@export var speed := 300.0
@export var jump_force := -400.0    # 음수! (위로 가야 하니까)
@export var gravity := 980.0

func _physics_process(delta):
    # 중력
    if not is_on_floor():
        velocity.y += gravity * delta

    # 점프 - 바닥에 있을 때만!
    if Input.is_action_just_pressed("jump") and is_on_floor():
        velocity.y = jump_force

    # 점프 높이 조절 (버튼을 빨리 떼면 낮게 점프)
    if Input.is_action_just_released("jump") and velocity.y < 0:
        velocity.y *= 0.5    # 상승 속도를 절반으로

    # 좌우 이동
    velocity.x = Input.get_axis("move_left", "move_right") * speed

    move_and_slide()
코요테 타임 (Coyote Time): 바닥을 벗어난 후 0.1초간 점프 허용 - 절벽 끝에서 떨어질 때 유예 시간 부여 
# 코요테 타임 구현
var coyote_timer := 0.0
const COYOTE_TIME := 0.1

func _physics_process(delta):
    if is_on_floor():
        coyote_timer = COYOTE_TIME
    else:
        coyote_timer -= delta

    if Input.is_action_just_pressed("jump") and coyote_timer > 0:
        velocity.y = jump_force
        coyote_timer = 0.0

완성: 플랫포머 컨트롤러

# player.gd - 플랫포머 캐릭터 컨트롤러 전체 코드
extends CharacterBody2D

# === 이동 파라미터 (인스펙터에서 조절 가능) ===
@export_group("Movement")
@export var speed := 300.0
@export var acceleration := 1500.0   # 가속도
@export var friction := 1200.0       # 마찰력

@export_group("Jump")
@export var jump_force := -420.0
@export var gravity := 980.0
@export var max_fall_speed := 600.0

# === 노드 참조 ===
@onready var sprite := $Sprite2D
@onready var anim := $AnimationPlayer

func _physics_process(delta: float) -> void:
    apply_gravity(delta)
    handle_jump()
    handle_movement(delta)
    update_animation()
    move_and_slide()

func apply_gravity(delta: float) -> void:
    if not is_on_floor():
        velocity.y += gravity * delta
        velocity.y = min(velocity.y, max_fall_speed)

func handle_jump() -> void:
    if Input.is_action_just_pressed("jump") and is_on_floor():
        velocity.y = jump_force
    # 가변 점프 높이
    if Input.is_action_just_released("jump") and velocity.y < 0:
        velocity.y *= 0.5

func handle_movement(delta: float) -> void:
    var dir := Input.get_axis("move_left", "move_right")
    if dir != 0:
        velocity.x = move_toward(velocity.x, dir * speed, acceleration * delta)
        sprite.flip_h = dir < 0
    else:
        velocity.x = move_toward(velocity.x, 0, friction * delta)

func update_animation() -> void:
    if not is_on_floor():
        anim.play("jump" if velocity.y < 0 else "fall")
    elif abs(velocity.x) > 10:
        anim.play("run")
    else:
        anim.play("idle")
