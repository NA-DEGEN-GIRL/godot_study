GPUParticles3D

GPU 파티클 대량의 3D 파티클을 GPU로 고속 처리

기본 설정

# GPUParticles3D 주요 프로퍼티
# amount: 파티클 수 (기본 8)
# lifetime: 수명 (초)
# one_shot: 일회성 발사
# preprocess: 미리 시뮬레이션 시간
# speed_scale: 속도 배율
# explosiveness: 동시 발사 비율 (0~1)
# randomness: 랜덤 비율
# fixed_fps: 고정 FPS (0이면 무제한)
# visibility_aabb: 가시 범위 바운딩 박스

# 코드로 파티클 제어
@onready var particles = $GPUParticles3D

func play_explosion():
    particles.one_shot = true
    particles.explosiveness = 1.0
    particles.amount = 100
    particles.emitting = true

func play_continuous():
    particles.one_shot = false
    particles.amount = 50
    particles.emitting = true

func stop():
    particles.emitting = false

씬 트리 구성

# 파티클 씬 구성
# FireEffect (GPUParticles3D)
#   amount: 200
#   lifetime: 1.5
#   process_material: ParticleProcessMaterial
#   draw_pass_1: QuadMesh (또는 SphereMesh)

# 파티클 생성 및 추가 (코드)
func create_spark_effect(pos: Vector3):
    var particles = GPUParticles3D.new()
    particles.amount = 30
    particles.lifetime = 0.5
    particles.one_shot = true
    particles.explosiveness = 0.9
    particles.emitting = true

    # 메시 설정
    var mesh = QuadMesh.new()
    mesh.size = Vector2(0.05, 0.05)
    particles.draw_pass_1 = mesh

    # 머티리얼 설정 (다음 슬라이드)
    var mat = ParticleProcessMaterial.new()
    particles.process_material = mat

    particles.global_position = pos
    add_child(particles)

    # 자동 삭제
    particles.finished.connect(
        particles.queue_free
    )

ParticleProcessMaterial 3D

파티클의 움직임, 크기, 색상을 세밀하게 제어

불꽃 이펙트

# 불꽃 파티클 머티리얼
func create_fire_material() -> ParticleProcessMaterial:
    var mat = ParticleProcessMaterial.new()

    # 방향과 확산
    mat.direction = Vector3(0, 1, 0)
    mat.spread = 15.0  # 확산 각도
    mat.initial_velocity_min = 2.0
    mat.initial_velocity_max = 4.0

    # 중력
    mat.gravity = Vector3(0, -0.5, 0)

    # 크기 변화 (커브)
    var size_curve = CurveTexture.new()
    var curve = Curve.new()
    curve.add_point(Vector2(0, 0.5))
    curve.add_point(Vector2(0.3, 1.0))
    curve.add_point(Vector2(1, 0))
    size_curve.curve = curve
    mat.scale_curve = size_curve
    mat.scale_min = 0.3
    mat.scale_max = 0.8

    # 색상 변화 (그라디언트)
    var gradient = GradientTexture1D.new()
    var grad = Gradient.new()
    grad.colors = [
        Color(1, 1, 0.3, 1),   # 노란 불꽃
        Color(1, 0.4, 0, 1),   # 주황
        Color(0.5, 0, 0, 0.5), # 어두운 빨강
        Color(0.2, 0.2, 0.2, 0) # 투명 연기
    ]
    gradient.gradient = grad
    mat.color_ramp = gradient

    return mat

주요 파라미터 정리

파라미터 설명
direction 파티클 방출 방향 (Vector3)
spread 확산 각도 (0~180도)
initial_velocity 초기 속도 (min/max)
gravity 중력 벡터
angular_velocity 회전 속도 (min/max)
linear_accel 선형 가속도
damping 감속 (공기저항)
scale_min/max 크기 범위
color_ramp 수명에 따른 색상 변화
scale_curve 수명에 따른 크기 변화
emission_shape 발사 형태 (점/구/박스 등)

3D 파티클 이미터 형태

파티클 생성 영역의 모양을 설정

이미터 형태 종류

형태 설명 활용 예시
Point 한 점에서 발사 스파크, 총구 화염
Sphere 구 표면/내부에서 발사 폭발, 힐링 오라
Sphere Surface 구 표면에서만 발사 쉴드 이펙트
Box 직육면체 영역에서 발사 비, 눈, 영역 효과
Ring 원형/도넛 영역에서 발사 마법진, 포탈
Points / Directed Points 정의된 점/방향에서 발사 메시 표면 효과

# 다양한 이미터 형태 설정
func setup_sphere_emitter(mat: ParticleProcessMaterial):
    mat.emission_shape = ParticleProcessMaterial.EMISSION_SHAPE_SPHERE
    mat.emission_sphere_radius = 2.0

func setup_box_emitter(mat: ParticleProcessMaterial):
    mat.emission_shape = ParticleProcessMaterial.EMISSION_SHAPE_BOX
    mat.emission_box_extents = Vector3(5, 0.1, 5)

func setup_ring_emitter(mat: ParticleProcessMaterial):
    mat.emission_shape = ParticleProcessMaterial.EMISSION_SHAPE_RING
    mat.emission_ring_radius = 3.0
    mat.emission_ring_inner_radius = 2.5
    mat.emission_ring_height = 0.1
    mat.emission_ring_axis = Vector3(0, 1, 0)

# 비 이펙트: Box 이미터 활용
func create_rain():
    var particles = GPUParticles3D.new()
    particles.amount = 500
    particles.lifetime = 2.0
    var mat = ParticleProcessMaterial.new()
    mat.emission_shape = ParticleProcessMaterial.EMISSION_SHAPE_BOX
    mat.emission_box_extents = Vector3(20, 0, 20)
    mat.gravity = Vector3(0, -15, 0)
    mat.direction = Vector3(0, -1, 0)
    mat.initial_velocity_min = 5.0
    mat.initial_velocity_max = 8.0
    particles.process_material = mat
    particles.position.y = 15.0
    add_child(particles)

마법 미사일 트레일 예시

# magic_missile_trail.gd
func create_magic_trail() -> GPUParticles3D:
    var p = GPUParticles3D.new()
    p.amount = 20
    p.lifetime = 0.8
    p.trail_enabled = true
    p.trail_lifetime = 0.4

    # 리본 메시
    var ribbon = RibbonTrailMesh.new()
    ribbon.size = 0.15
    ribbon.sections = 6
    p.draw_pass_1 = ribbon

    # 머티리얼
    var mat = ParticleProcessMaterial.new()
    mat.direction = Vector3(0, 0, -1)
    mat.initial_velocity_min = 0.5
    mat.initial_velocity_max = 1.0
    mat.gravity = Vector3.ZERO

    # 색상: 밝은 파랑 -> 투명
    var grad_tex = GradientTexture1D.new()
    var grad = Gradient.new()
    grad.colors = [
        Color(0.3, 0.7, 1, 1),
        Color(0.1, 0.3, 1, 0)
    ]
    grad_tex.gradient = grad
    mat.color_ramp = grad_tex

    p.process_material = mat
    return p
Tip: trail_lifetime은 파티클 lifetime보다 짧게 설정하세요. 너무 길면 GPU 부하가 커질 수 있습니다.

GPUParticlesAttractor3D

파티클 흡인 파티클을 끌어당기거나 밀어내는 힘

어트랙터 종류

노드 영향 범위 용도
GPUParticlesAttractorSphere3D 구형 영역 블랙홀, 아이템 흡수
GPUParticlesAttractorBox3D 박스 영역 바람 통로, 환기구
GPUParticlesAttractorVectorField3D 벡터 필드 복잡한 힘의 장

strength 양수 = 끌어당김, 음수 = 밀어냄
attenuation 거리에 따른 힘 감쇠 커브
cull_mask 영향 받을 파티클 레이어

코드 예시

# 아이템 흡수 효과
func create_absorb_effect(target_pos: Vector3):
    # 어트랙터 생성
    var attractor = GPUParticlesAttractorSphere3D.new()
    attractor.radius = 3.0
    attractor.strength = 5.0
    attractor.attenuation = 1.5
    attractor.global_position = target_pos
    add_child(attractor)

    # 일정 시간 후 제거
    await get_tree().create_timer(2.0).timeout
    attractor.queue_free()

# 바람 효과 (밀어내기)
func create_wind_zone(pos: Vector3, size: Vector3):

GPUParticlesCollision3D

파티클 충돌 파티클이 환경 표면과 상호작용

충돌 노드 종류

노드 방식 특징
GPUParticlesCollisionSphere3D 구형 충돌 빠름, 단순한 형태
GPUParticlesCollisionBox3D 박스 충돌 평면/벽에 적합
GPUParticlesCollisionSDF3D SDF 베이킹 정밀, 복잡한 형태

SDF 충돌 설정

# 씬 트리 구성
# Level (Node3D)
#   +-- GPUParticlesCollisionSDF3D
#   |     size: (20, 10, 20)
#   |     resolution: 64
#   |     (Bake SDF 버튼 클릭)
#   +-- GPUParticles3D (불꽃)
#   +-- MeshInstance3D (바닥)

# ParticleProcessMaterial 충돌 설정
func setup_collision_material(mat: ParticleProcessMaterial):
    # 충돌 모드
    mat.collision_mode = ParticleProcessMaterial.COLLISION_RIGID
    # 반발 계수 (0=정지, 1=완전 반발)
    mat.collision_bounce = 0.3
    # 마찰 (0=미끄러움, 1=즉시 정지)
    mat.collision_friction = 0.5
    # 충돌 시 흡수 (파티클 제거 확률)
    mat.collision_use_scale = true

# HeightField: 실시간 지형 충돌
# GPUParticlesCollisionHeightField3D
#   follow_camera_enabled: true
#   (카메라 주변 자동 업데이트)
권장 조합: 실내 레벨에는 SDF, 넓은 야외 지형에는 HeightField, 단순한 상호작용에는 Sphere/Box를 사용하세요.

CPUParticles3D vs GPUParticles3D

상황에 맞는 파티클 시스템 선택

비교 항목 GPUParticles3D CPUParticles3D
처리 위치 GPU (병렬 처리) CPU (순차 처리)
최대 파티클 수 수십만 개 가능 수천 개 정도 권장
개별 파티클 접근 불가능 가능
어트랙터/충돌 전용 노드 지원 수동 구현 필요
트레일 지원 미지원
호환성 Forward+/Mobile 최적 모든 렌더러에서 동작
적합한 용도 대량 파티클, 시각 효과 소량, 정밀 제어, 저사양

Decal 노드

표면 효과 3D 표면에 텍스처를 투영

Decal 주요 속성

# Decal 인스펙터 설정
# size: Vector3(1, 2, 1)  - 투영 범위
# texture_albedo: 알베도 텍스처
# texture_normal: 노멀맵 텍스처
# texture_orm: ORM(AO/Roughness/Metallic)
# texture_emission: 이미시브 텍스처
# modulate: Color(1,1,1,1) - 색상 조절
# albedo_mix: 0.0~1.0 - 혼합 강도
# upper_fade: 위쪽 페이드 (0~1)
# lower_fade: 아래쪽 페이드 (0~1)
# normal_fade: 노멀 기반 페이드
# cull_mask: 적용 대상 레이어

Decal의 size.y는 투영 깊이입니다. 표면에 가까운 오브젝트만 영향받으므로 적절한 깊이를 설정하세요.

동적 데칼 시스템

# bullet_decal.gd - 총알 자국 시스템
extends Node3D

@export var decal_texture: Texture2D
@export var max_decals: int = 50
var decal_pool: Array[Decal] = []
var current_index: int = 0

func _ready():
    for i in max_decals:
        var decal = Decal.new()
        decal.size = Vector3(0.3, 0.5, 0.3)
        decal.texture_albedo = decal_texture
        decal.modulate = Color(1, 1, 1, 0.8)
        decal.upper_fade = 0.3
        decal.lower_fade = 0.3
        decal.visible = false
        add_child(decal)
        decal_pool.append(decal)

func spawn_decal(pos: Vector3, normal: Vector3):
    var decal = decal_pool[current_index]
    decal.global_position = pos + normal * 0.01
    # 노멀 방향으로 데칼 정렬
    if normal != Vector3.UP:
        decal.look_at(pos + normal, Vector3.UP)
    decal.visible = true
    current_index = (current_index + 1) % max_decals

    # 페이드 아웃 (선택)
    var tween = create_tween()
    tween.tween_property(decal, "modulate:a", 0.0, 10.0)

FogVolume

볼류메트릭 안개 부피감 있는 3D 안개 효과

FogVolume 설정

# 씬 트리 구성
# Level (Node3D)
#   +-- WorldEnvironment
#   |     Environment:
#   |       volumetric_fog_enabled: true
#   |       volumetric_fog_density: 0.02
#   |       volumetric_fog_albedo: Color(0.8,0.8,0.9)
#   +-- FogVolume (동굴 안개)
#   |     size: Vector3(10, 3, 10)
#   |     shape: Box
#   |     material: FogMaterial
#   +-- FogVolume2 (늪 안개)
#         size: Vector3(20, 1, 20)
#         shape: Ellipsoid

# FogMaterial 속성
# density: 0.5  (안개 밀도)
# albedo: Color(0.7, 0.7, 0.8)
# emission: Color(0, 0, 0) (자체 발광)
# height_falloff: 1.0 (높이 감쇠)

코드로 안개 제어

# 동적 안개 밀도 변화
@onready var fog_vol = $FogVolume
@onready var env = $WorldEnvironment.environment

func enter_cave():
    var tween = create_tween()
    # FogVolume 밀도 증가
    var mat = fog_vol.material as FogMaterial
    tween.tween_property(mat, "density", 1.0, 2.0)
    # 전역 안개도 진하게
    tween.parallel().tween_property(
        env, "volumetric_fog_density", 0.08, 2.0
    )

func exit_cave():
    var tween = create_tween()
    var mat = fog_vol.material as FogMaterial
    tween.tween_property(mat, "density", 0.1, 2.0)
    tween.parallel().tween_property(
        env, "volumetric_fog_density", 0.01, 2.0
    )
성능: Volumetric Fog는 비용이 큽니다. 프로젝트 설정의 volumetric_fog_quality를 Low로 설정하거나, FogVolume 수를 최소화하세요.

LOD (Level of Detail)

최적화 거리에 따라 메시 디테일을 자동 조절

자동 LOD (메시 임포트)

# 메시 임포트 설정 (Advanced Import Settings)
# Meshes > LOD:
#   lod_mode: "Generate"
#   normal_merge_angle: 60
#   normal_split_angle: 25

# MeshInstance3D 인스펙터
# lod_bias: 1.0  (기본값)
#   값이 작을수록 빨리 저해상도로 전환
#   0.5 = 기본 거리의 절반에서 전환
#   2.0 = 기본 거리의 두 배에서 전환

# 프로젝트 설정
# Rendering > Mesh LOD
# lod_threshold: 1.0 (전체 LOD 감도)
# mesh_lod_threshold: 1.0

glTF나 FBX 파일을 임포트할 때 LOD Generate를 활성화하면 Godot이 자동으로 여러 단계의 간소화된 메시를 생성합니다.

수동 LOD 설정

# 수동 LOD: 여러 메시를 직접 관리
# Tree (Node3D)
#   +-- TreeHigh (MeshInstance3D)  [5000 poly]
#   +-- TreeMed (MeshInstance3D)   [1000 poly]
#   +-- TreeLow (MeshInstance3D)   [200 poly]
#   +-- TreeBillboard (Sprite3D)   [2 poly]

extends Node3D

@onready var meshes = [$TreeHigh, $TreeMed, $TreeLow, $TreeBillboard]
var lod_distances = [0, 20, 50, 100]

func _process(delta):
    var cam = get_viewport().get_camera_3d()
    if not cam:
        return
    var dist = global_position.distance_to(
        cam.global_position
    )
    for i in meshes.size():
        var next_dist = lod_distances[i + 1] if i + 1 < lod_distances.size() else INF
        meshes[i].visible = (dist >= lod_distances[i] and dist < next_dist)
Tip: 가능하면 Godot의 자동 LOD를 사용하세요. 수동 LOD는 빌보드나 특수한 경우에만 필요합니다.

HLOD (Hierarchical LOD)

대규모 최적화 먼 거리의 오브젝트 그룹을 하나로 병합

HLOD 개념

LOD vs HLOD

  • LOD: 개별 오브젝트의 폴리곤 수 감소
  • HLOD: 오브젝트 그룹을 하나의 메시로 병합
  • 효과: draw call 수를 극적으로 감소
# HLOD 수동 구현 개념
# Village (Node3D)
#   +-- VillageLOD (MeshInstance3D)

# hlod_group.gd - 간단한 HLOD 관리
extends Node3D

@export var proxy_mesh: MeshInstance3D
@export var detail_group: Node3D
@export var switch_distance: float = 100.0
@export var fade_range: float = 20.0

func _process(delta):
    var cam = get_viewport().get_camera_3d()
    if not cam:
        return
    var dist = global_position.distance_to(
        cam.global_position
    )

    if dist > switch_distance + fade_range:
        # 먼 거리: 프록시 메시만 표시
        proxy_mesh.visible = true
        detail_group.visible = false
    elif dist < switch_distance:
        # 가까운 거리: 개별 메시 표시
        proxy_mesh.visible = false
        detail_group.visible = true
    else:
        # 전환 구간: 둘 다 표시 (크로스페이드)
        proxy_mesh.visible = true
        detail_group.visible = true
        var t = (dist - switch_distance) / fade_range
        proxy_mesh.transparency = 1.0 - t
Tip: HLOD 프록시 메시는 Blender 등에서 Decimate 도구로 만들거나, MeshInstance3D를 병합한 뒤 단순화합니다.

Visibility Range

거리 기반 표시 카메라 거리에 따른 자동 표시/숨김

설정 방법

# MeshInstance3D 인스펙터
# Visibility Range:
#   Begin: 0      (이 거리부터 보임)
#   Begin Margin: 1  (페이드 인 범위)
#   End: 50       (이 거리에서 사라짐)
#   End Margin: 5    (페이드 아웃 범위)
#   Fade Mode: Dependencies

# LOD 체인 구성 예시:
# Rock (Node3D)
#   +-- RockHigh (MeshInstance3D)
#         begin: 0, end: 30
#   +-- RockMed (MeshInstance3D)
#         begin: 25, end: 60
#   +-- RockLow (MeshInstance3D)
#         begin: 55, end: 120

Fade Mode

  • Disabled: 즉시 전환 (팝핑 발생)
  • Self: 자체 투명도로 페이드
  • Dependencies: 연결된 LOD와 교차 페이드

코드로 설정

# visibility_range 코드 설정
func setup_visibility_range(mesh: MeshInstance3D,
    begin: float, end: float):
    mesh.visibility_range_begin = begin
    mesh.visibility_range_begin_margin = begin * 0.1
    mesh.visibility_range_end = end
    mesh.visibility_range_end_margin = end * 0.1
    mesh.visibility_range_fade_mode = GeometryInstance3D.VISIBILITY_RANGE_FADE_DEPENDENCIES

# 대량의 풀/나무에 적용
func setup_vegetation():
    for grass in get_tree().get_nodes_in_group("grass"):
        setup_visibility_range(grass, 0, 30)
    for tree in get_tree().get_nodes_in_group("trees"):
        setup_visibility_range(tree, 0, 100)
    for rock in get_tree().get_nodes_in_group("rocks"):
        setup_visibility_range(rock, 0, 80)
Tip: Begin과 End 사이에 Margin 만큼의 페이드 구간이 생깁니다. Margin을 너무 크게 하면 투명 오브젝트가 많아져 오히려 성능이 떨어질 수 있습니다.

        var pos = Vector3(
            randf_range(-area_size, area_size),
            0,
            randf_range(-area_size, area_size)
        )
        # 랜덤 회전과 크기
        var t = Transform3D.IDENTITY
        t = t.rotated(Vector3.UP, randf() * TAU)
        var s = randf_range(0.8, 1.2)
        t = t.scaled(Vector3(s, s, s))
        t.origin = pos
        mm.set_instance_transform(i, t)

        # 약간씩 다른 색상
        var green = randf_range(0.4, 0.8)
        mm.set_instance_color(i, Color(0.2, green, 0.1))

    multimesh = mm

Draw Call 줄이기

성능 핵심 GPU 명령 호출 횟수 최소화

최적화 전략

Draw Call 감소 기법

  1. 머티리얼 공유 - 동일 머티리얼을 여러 메시에 재사용
  2. 메시 병합 - 정적 오브젝트를 하나의 메시로 합침
  3. 텍스처 아틀라스 - 여러 텍스처를 하나의 큰 텍스처에 합침
  4. MultiMesh - 반복 오브젝트 인스턴싱
  5. LOD/HLOD - 먼 거리 오브젝트 단순화/병합
  6. Visibility Range - 보이지 않는 오브젝트 제외

메시 병합 코드

# 정적 오브젝트 메시 병합
func merge_static_meshes(parent: Node3D):
    var surfaces_tool = SurfaceTool.new()
    surfaces_tool.begin(Mesh.PRIMITIVE_TRIANGLES)

    for child in parent.get_children():
        if child is MeshInstance3D:
            var mesh = child.mesh
            for surf_idx in mesh.get_surface_count():
                surfaces_tool.append_from(
                    mesh, surf_idx, child.transform
                )

    surfaces_tool.generate_normals()
    var merged_mesh = surfaces_tool.commit()

    # 병합된 메시 노드 생성
    var merged = MeshInstance3D.new()
    merged.mesh = merged_mesh
    parent.get_parent().add_child(merged)

    # 원본 제거
    parent.queue_free()
주의: 메시 병합은 정적(움직이지 않는) 오브젝트에만 적용하세요. 병합하면 개별 오브젝트의 컬링이 불가능해지므로 너무 넓은 범위를 병합하면 오히려 역효과입니다.
