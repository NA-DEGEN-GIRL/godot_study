3D 물리 개요

Godot 4의 3D 물리 엔진

GodotPhysics3D (기본) 또는 Jolt Physics (확장) 중 선택 가능. 프로젝트 설정에서 변경합니다.

엔진 비교

항목GodotPhysics3DJolt
설치기본 내장별도 활성화
정확도보통높음
성능보통우수

기본 설정

# 프로젝트 설정 (project.godot)
# Physics > 3D > Physics Engine:
#   GodotPhysics3D (기본)
#   Jolt (4.4+에서 내장 옵션)

# 기본 중력
# physics/3d/default_gravity = 9.8
# physics/3d/default_gravity_vector
#   = (0, -1, 0)

# 물리 프레임
# physics/common/physics_ticks_per_second
#   = 60
Jolt 활성화: Godot 4.4 이상에서는 프로젝트 설정 Physics > 3D > Physics Engine에서 Jolt Physics를 직접 선택할 수 있습니다. 더 안정적인 물리 시뮬레이션이 필요하면 Jolt를 추천합니다.

StaticBody3D

고정 물체 물리 세계에서 절대 움직이지 않는 3D 바디

씬 트리 구조

# 바닥 구성
Ground (StaticBody3D)
  +-- MeshInstance3D (시각적 표현)
  |     mesh = PlaneMesh (20x20)
  +-- CollisionShape3D (물리 충돌)
        shape = WorldBoundaryShape3D

# 벽 구성
Wall (StaticBody3D)
  +-- MeshInstance3D
  |     mesh = BoxMesh (10x3x0.5)
  +-- CollisionShape3D
        shape = BoxShape3D

코드로 생성

# 바닥 만들기
func create_floor():
    var body = StaticBody3D.new()

    # 시각적 메시
    var mesh_inst = MeshInstance3D.new()
    var plane = PlaneMesh.new()
    plane.size = Vector2(20, 20)
    mesh_inst.mesh = plane
    body.add_child(mesh_inst)

    # 충돌 형태
    var col = CollisionShape3D.new()
    col.shape = WorldBoundaryShape3D.new()
    body.add_child(col)

    add_child(body)
주의: MeshInstance3D만 추가하면 눈에는 보이지만 물리 충돌이 없습니다. 반드시 CollisionShape3D를 함께 추가하세요. 메시 선택 후 상단 메뉴 Mesh > Create Trimesh/Convex Collision을 사용하면 자동 생성됩니다.

기본 이동 코드

extends CharacterBody3D

const SPEED = 5.0
const GRAVITY = 9.8
const JUMP_FORCE = 4.5

func _physics_process(delta):
    # 중력 적용
    if not is_on_floor():
        velocity.y -= GRAVITY * delta

    # 점프
    if Input.is_action_just_pressed("jump"):
        if is_on_floor():
            velocity.y = JUMP_FORCE

    # 이동 입력
    var input = Input.get_vector(
        "left", "right", "forward", "back"
    )
    var direction = Vector3(
        input.x, 0, input.y
    ).normalized()

    velocity.x = direction.x * SPEED
    velocity.z = direction.z * SPEED

    move_and_slide()

카메라 기준 이동

# 카메라 방향 기준 이동 (3인칭)
extends CharacterBody3D

@onready var camera = $"../Camera3D"

func get_camera_direction() -> Vector3:
    var input = Input.get_vector(
        "left", "right", "forward", "back"
    )

    # 카메라의 앞/오른쪽 방향
    var cam_forward = -camera.basis.z
    var cam_right = camera.basis.x

    # Y축 제거 (바닥 평면만)
    cam_forward.y = 0
    cam_right.y = 0
    cam_forward = cam_forward.normalized()
    cam_right = cam_right.normalized()

    # 카메라 기준 방향으로 변환
    return (cam_right * input.x + \
        cam_forward * input.y).normalized()

body_entered / body_exited (3D)

Area3D 시그널 물리 바디의 3D 영역 진입/이탈 감지

기본 사용

# 데미지 존 (용암, 함정)
extends Area3D

@export var damage_per_second: float = 20.0
var bodies_inside: Array[Node3D] = []

func _ready():
    body_entered.connect(_on_body_entered)
    body_exited.connect(_on_body_exited)

func _on_body_entered(body: Node3D):
    if body.is_in_group("player"):
        bodies_inside.append(body)

func _on_body_exited(body: Node3D):
    bodies_inside.erase(body)

func _physics_process(delta):
    for body in bodies_inside:
        if body.has_method("take_damage"):
            body.take_damage(
                damage_per_second * delta
            )

RayCast3D

광선 감지 3D 공간에서 광선을 쏘아 첫 번째 충돌을 감지

extends CharacterBody3D

@onready var ray = $RayCast3D

func _ready():
    # 앞쪽으로 10유닛 광선
    ray.target_position = Vector3(0, 0, -10)
    ray.collision_mask = 0b00000110  # 레이어 2,3 감지

func _physics_process(delta):
    if ray.is_colliding():
        var collider = ray.get_collider()
        var point = ray.get_collision_point()
        var normal = ray.get_collision_normal()

        if collider.is_in_group("enemy"):
            print("적 발견: %s (거리: %.1f)" % [
                collider.name,
                global_position.distance_to(point)
            ])

ShapeCast3D

도형 기반 감지 3D 도형을 밀어내며 경로 상 모든 충돌을 감지

RayCast3D vs ShapeCast3D

구분RayCast3DShapeCast3D
감지 범위가느다란 선도형 크기만큼
결과 수1개여러 개
성능빠름보통
용도시선, 조준, 히트스캔범위 감지, 이동 검사
# 폭발 범위 감지
extends Node3D

@onready var cast = $ShapeCast3D

func explode():
    # 구형 범위로 감지
    cast.shape = SphereShape3D.new()
    cast.shape.radius = 5.0
    cast.target_position = Vector3.ZERO
    cast.force_shapecast_update()

    var count = cast.get_collision_count()
    for i in count:
        var obj = cast.get_collider(i)
        if obj.has_method("take_damage"):
            var dist = global_position.\
                distance_to(obj.global_position)
            var dmg = 100 * (1.0 - dist / 5.0)
            obj.take_damage(dmg)
활용: 캐릭터 크기의 ShapeCast를 이동 방향으로 쏘면 좁은 통로를 지나갈 수 있는지 미리 검사할 수 있습니다. 3인칭 카메라의 벽 관통 방지에도 유용합니다.

var time: float = 0.0

func _ready():
    base_y = position.y
    body_entered.connect(_on_body_entered)
    # 자석 범위 설정
    $CollisionShape3D.shape.radius = magnet_range

func _on_body_entered(body: Node3D):
    if body.is_in_group("player") and not is_collected:
        target_player = body

func _process(delta):
    # 회전 + 떠다니는 연출
    time += delta
    $MeshInstance3D.rotation.y += spin_speed * delta
    if target_player == null:
        position.y = base_y + sin(time * 3.0) * bob_height

func _physics_process(delta):
    if target_player and not is_collected:
        # 자석처럼 플레이어에게 끌려감
        var dir = (target_player.global_position - global_position)
        var dist = dir.length()

        if dist < collect_range:
            collect()
        else:
            global_position += dir.normalized() * magnet_speed * delta

func collect():
    is_collected = true
    if target_player.has_method("add_score"):
        target_player.add_score(point_value)
    # 수집 연출
    var tween = create_tween()
    tween.tween_property(self, "scale", Vector3.ZERO, 0.2)
    tween.tween_callback(queue_free)
