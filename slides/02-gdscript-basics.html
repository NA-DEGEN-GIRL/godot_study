GDScript 개요

언어 특징

  • Godot 전용 스크립트 언어
  • 들여쓰기 기반 (탭 사용 권장)
  • 세미콜론, 중괄호 불필요
  • 동적 타이핑 + 선택적 정적 타이핑
  • Godot API와 깊은 통합
# GDScript 기본 예시
extends Node2D

var score: int = 0

func _ready():
    # 들여쓰기가 코드 블록을 결정
    print("게임 시작!")
    score += 10
    if score > 5:
        print("점수: ", score)
주의: GDScript는 탭(Tab)과 스페이스를 혼용하면 오류가 발생합니다. Godot 에디터는 기본적으로 탭을 사용합니다.

변수 선언: var & const

var - 변수

# 기본 변수 선언
var health = 100
var player_name = "Hero"
var is_alive = true
var speed = 5.5

# 초기화 없이 선언 (null)
var target

const - 상수

# 상수는 변경 불가!
const MAX_HEALTH = 100
const GRAVITY = 980.0
const GAME_TITLE = "My Game"
const TILE_SIZE = 64

# 이렇게 하면 오류 발생:
# MAX_HEALTH = 200  # Error!

네이밍 컨벤션

  • 변수/함수: snake_case (예: player_health, move_speed)
  • 상수: UPPER_SNAKE_CASE (예: MAX_SPEED, TILE_SIZE)
  • 클래스: PascalCase (예: PlayerController, EnemyAI)

기본 자료형

자료형 설명 예시
int 정수 42, -10, 0xFF, 0b1010
float 부동소수점 3.14, -0.5, 1e10
String 문자열 "Hello", 'World'
bool 참/거짓 true, false
null 값 없음 null 
# Godot 전용 타입
var pos: Vector2 = Vector2(100, 200)
var pos3d: Vector3 = Vector3(1, 2, 3)

var path: NodePath = ^"Player/Sprite2D"

타입 힌트 (Type Hints)

타입 힌트 없이

# 동적 타이핑
var health = 100
var name = "Hero"
var pos = Vector2(0, 0)

func get_damage(amount):
    health -= amount
    return health

타입 힌트 사용

# 정적 타이핑 (권장!)
var health: int = 100
var name: String = "Hero"
var pos: Vector2 = Vector2.ZERO

func get_damage(amount: int) -> int:
    health -= amount
    return health

타입 힌트의 장점

  • 자동완성 에디터가 타입을 알아 정확한 제안 제공
  • 오류 감지 실행 전에 타입 불일치 경고
  • 성능 향상 런타임 타입 체크 불필요
Tip: := 연산자로 타입을 자동 추론할 수 있습니다. var speed := 5.0은 자동으로 float 타입이 됩니다.

문자열 처리 (String)

# 문자열 기본
var greeting: String = "안녕하세요"
var name: String = 'Godot'

# 문자열 결합
var message = greeting + ", " + name + "!"

# 문자열 포맷팅 (% 연산자)
var formatted = "체력: %d / %d" % [75, 100]
var info = "이름: %s, 레벨: %d" % ["Hero", 5]

# 문자열 메서드
var text = "  Hello World  "
print(text.strip_edges())    # "Hello World"
print(text.to_upper())       # "  HELLO WORLD  "
print(text.length())         # 15
print("Hello".begins_with("He"))  # true
print("World".find("orl"))        # 1
Tip: 여러 줄 문자열은 """삼중 따옴표"""를 사용합니다. JSON이나 긴 텍스트에 유용합니다.

배열 (Array)

# 배열 생성
var items: Array = [1, 2, 3]
var names = ["Alice", "Bob"]
var mixed = [1, "hello", true]

# 타입 지정 배열 (Godot 4)
var scores: Array[int] = [90, 85, 77]
var enemies: Array[Node2D] = []

# 요소 접근
print(items[0])   # 1
print(items[-1])  # 3 (마지막)
# 배열 메서드
items.append(4)       # 추가
items.insert(0, 0)    # 위치 삽입
items.remove_at(1)    # 인덱스 삭제
items.erase(3)        # 값으로 삭제
items.pop_back()      # 마지막 꺼냄
items.pop_front()     # 첫 번째 꺼냄

# 검색 및 정렬
items.find(2)         # 인덱스 반환
items.has(5)          # 포함 여부
items.sort()          # 정렬
items.reverse()       # 역순
items.shuffle()       # 셔플
print(items.size())   # 크기
주의: 배열 인덱스는 0부터 시작합니다. 존재하지 않는 인덱스에 접근하면 런타임 오류가 발생합니다.

딕셔너리 (Dictionary)

# 딕셔너리 생성
var player: Dictionary = {
    "name": "Hero",
    "level": 5,
    "hp": 100,
    "items": ["sword", "shield"]
}

# 값 접근
print(player["name"])      # "Hero"
print(player.get("hp"))    # 100
print(player.get("mp", 0)) # 0 (기본값)

# 값 수정/추가
player["hp"] = 80
player["mp"] = 50          # 새 키 추가

# 유용한 메서드
print(player.has("name"))  # true
print(player.keys())       # 모든 키 배열
print(player.values())     # 모든 값 배열
print(player.size())       # 5
player.erase("mp")         # 키 삭제
Tip: 딕셔너리는 JSON과 호환됩니다. JSON.stringify(dict)와 JSON.parse_string(json_str)으로 변환 가능합니다.

연산자

산술 연산자

# 산술 연산
print(10 + 3)   # 13
print(10 - 3)   # 7
print(10 * 3)   # 30
print(10 / 3)   # 3 (int / int)
print(10.0 / 3) # 3.333...
print(10 % 3)   # 1 (나머지)

비교 & 논리 연산자

print(5 == 5)    # true
print(5 != 3)    # true
print(5 > 3)     # true
print(5 <= 5)    # true

# 논리 연산
print(true and false) # false
print(true or false)  # true
print(not true)       # false

# in 연산자
print("a" in "abc")   # true
print(3 in [1, 2, 3]) # true

+= -= *= /= %= 모두 지원합니다. 예: health -= damage

조건문: if / elif / else

# 기본 조건문
var hp: int = 30

if hp <= 0:
    print("사망!")
elif hp < 30:
    print("위험!")
elif hp < 70:
    print("주의")
else:
    print("양호")

# 한 줄 조건문 (삼항)
var status = "살아있음" if hp > 0 else "사망"

# 중첩 조건
if is_alive:
    if has_weapon:
        attack()
    else:
        flee()

match 문 (switch 대체)

# match 문 - 패턴 매칭
var state = "idle"

match state:
    "idle":
        print("대기 중")
    "run":
        print("달리는 중")
    "jump", "fall":
        print("공중에 있음")
    _:
        print("알 수 없는 상태")

# match로 타입 패턴 매칭
match typeof(value):
    TYPE_INT:
        print("정수입니다")
    TYPE_STRING:
        print("문자열입니다")

match 문 심화

# 배열 패턴 매칭
var command = ["move", 10, 20]

match command:
    ["move", var x, var y]:
        print("이동: ", x, ", ", y)
    ["attack", var target]:
        print("공격: ", target)
    ["heal"]:
        print("회복")
    _:
        print("알 수 없는 명령")

# 딕셔너리 패턴 매칭
var event = {"type": "damage", "amount": 50}

match event:
    {"type": "damage", "amount": var dmg}:
        print("피해: ", dmg)
    {"type": "heal", "amount": var heal}:
        print("회복: ", heal)

match vs if-elif

  • match: 하나의 값에 대한 여러 패턴 비교 시 가독성 우수
  • if-elif: 범위 비교나 복잡한 조건에 적합

반복문: for & while

for 반복문

# range로 반복
for i in range(5):
    print(i)  # 0, 1, 2, 3, 4

for i in range(2, 8, 2):
    print(i)  # 2, 4, 6

# 배열 순회
var items = ["검", "방패", "물약"]
for item in items:
    print(item)

# 딕셔너리 순회
var stats = {"hp": 100, "mp": 50}
for key in stats:
    print(key, ": ", stats[key])

while 반복문

# while 기본
var count: int = 0
while count < 5:
    print(count)
    count += 1

# break / continue
for i in range(10):
    if i == 3:
        continue  # 3 건너뛰기
    if i == 7:
        break     # 7에서 중단
    print(i)
# 출력: 0, 1, 2, 4, 5, 6
주의: while 문에서 종료 조건을 빠뜨리면 무한 루프에 빠집니다! 게임이 멈추므로 항상 탈출 조건을 확인하세요.

함수 정의 (func)

# 기본 함수
func greet():
    print("안녕하세요!")

# 매개변수와 반환값
func add(a: int, b: int) -> int:
    return a + b

# 기본값 매개변수
func create_enemy(name: String = "Slime", hp: int = 50) -> Dictionary:
    return {"name": name, "hp": hp}

# 호출
greet()                         # "안녕하세요!"
var result = add(3, 5)          # 8
var enemy1 = create_enemy()     # {"name": "Slime", "hp": 50}
var enemy2 = create_enemy("Dragon", 500)

Godot 콜백 함수 (언더스코어 시작)

  • _ready() - 초기화 시 호출
  • _process(delta) - 매 프레임 호출
  • _physics_process(delta) - 물리 프레임마다 호출
  • _input(event) - 입력 이벤트 발생 시 호출

함수 심화: 매개변수 & 반환

다중 반환값

# 배열로 다중 반환
func get_min_max(arr: Array) -> Array:
    return [arr.min(), arr.max()]

var result = get_min_max([3, 1, 7])
print(result[0])  # 1
print(result[1])  # 7

# 딕셔너리로 반환
func get_stats() -> Dictionary:
    return {
        "hp": 100,
        "mp": 50,
        "atk": 25
    }

Callable & 람다

# 함수를 변수에 저장
var my_func: Callable = greet
my_func.call()  # "안녕하세요!"

# 람다 (익명 함수)
var double = func(x): return x * 2
print(double.call(5))  # 10

# 배열 정렬에 활용
var items = [
    {"name": "B", "price": 30},
    {"name": "A", "price": 10},
]
items.sort_custom(func(a, b):
    return a["price"] < b["price"]
)

주요 내장 함수

분류함수설명
수학 abs(), ceil(), floor(), round() 절대값, 올림, 내림, 반올림
범위 clamp(val, min, max) 값을 min~max 사이로 제한
보간 lerp(from, to, weight) 두 값 사이를 보간
난수 randi(), randf(), randi_range() 랜덤 정수/실수 생성
변환 str(), int(), float() 타입 변환
출력 print(), push_error() 콘솔 출력, 에러 출력 
# 자주 쓰는 내장 함수 예시
var hp = clamp(hp - damage, 0, MAX_HP)   # 0~MAX_HP 범위 제한
var smooth_pos = lerp(pos, target, 0.1)  # 부드러운 이동
var dice = randi_range(1, 6)             # 주사위 (1~6)
var angle = deg_to_rad(90)               # 각도 변환

클래스 기초

기본 클래스 구조

# player.gd
class_name Player
extends CharacterBody2D

# 멤버 변수
var hp: int = 100
var attack_power: int = 10

# 생성 시 호출
func _ready():
    print(name, " 생성됨!")

# 멤버 함수
func take_damage(amount: int):
    hp -= amount
    if hp <= 0:
        die()

func die():
    queue_free()

내부(Inner) 클래스

# game.gd
extends Node

# 내부 클래스 정의
class Inventory:
    var items: Array[String] = []

    func add(item: String):
        items.append(item)

    func remove(item: String):
        items.erase(item)

    func has(item: String) -> bool:
        return item in items

# 사용
var bag = Inventory.new()
bag.add("검")
bag.add("물약")

열거형 (enum)

# enum 정의
enum State { IDLE, RUN, JUMP, FALL, ATTACK }
enum Direction { UP, DOWN, LEFT, RIGHT }

# 값은 자동으로 0부터 증가
# State.IDLE == 0, State.RUN == 1, ...

# 커스텀 값 지정
enum Element {
    FIRE = 10,
    WATER = 20,
    EARTH = 30,
    WIND = 40
}

# 사용 예시
var current_state: State = State.IDLE

func update_state():
    match current_state:
        State.IDLE:
            if Input.is_action_pressed("move"):
                current_state = State.RUN
        State.RUN:
            if not is_on_floor():
                current_state = State.FALL
        State.JUMP:
            if velocity.y > 0:
                current_state = State.FALL
Tip: enum은 match 문과 함께 사용하면 상태 머신(State Machine) 패턴을 깔끔하게 구현할 수 있습니다.

static 변수 & 함수

static 변수

class_name GameConfig

static 함수

# 유틸리티 클래스
class_name MathUtils

static func distance_2d(
    a: Vector2, b: Vector2
) -> float:
    return a.distance_to(b)

static func random_dir() -> Vector2:
    var angle = randf() * TAU
    return Vector2(
        cos(angle), sin(angle)
    )
주의: static 함수 안에서는 self나 인스턴스 멤버 변수에 접근할 수 없습니다. 오직 static 멤버와 매개변수만 사용 가능합니다.
