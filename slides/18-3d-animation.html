3D 모델 임포트

외부 3D 모델링 툴에서 만든 에셋을 Godot으로 가져오기

포맷 확장자 장점 단점
glTF 2.0 (권장) .gltf / .glb 오픈 표준, 메시+애니메이션+스켈레톤 포함, PBR 지원 일부 복잡한 리깅 제한
FBX .fbx 업계 표준, 복잡한 애니메이션 지원 독점 포맷, ufbx 라이브러리 필요
OBJ .obj / .mtl 단순, 널리 지원 애니메이션/스켈레톤 미지원
Blend .blend Blender 직접 임포트 (Blender 설치 필요) Blender가 시스템에 설치되어 있어야 함

임포트 절차

  1. 3D 모델 파일을 res://assets/models/에 배치
  2. Godot이 자동으로 .import 파일 생성
  3. FileSystem 독에서 더블 클릭하여 씬으로 열기
  4. 필요시 임포트 설정 조정 후 Reimport

Skeleton3D

뼈대 3D 캐릭터의 본(Bone) 계층 구조

Skeleton3D 구조

  • 본(Bone)들의 부모-자식 계층
  • 각 본은 인덱스(int)로 식별
  • 본 이름으로 인덱스 검색 가능
  • Rest Pose = 기본 포즈 (T-Pose 등)
  • 각 본의 로컬/글로벌 트랜스폼 제어 가능
# 일반적인 인체 본 계층 구조
# Root
# └ Hips
#   ├ Spine
#   │ ├ Chest
#   │ │ ├ Neck > Head
#   │ │ ├ LeftShoulder > LeftArm > LeftForeArm > LeftHand
#   │ │ └ RightShoulder > RightArm > RightForeArm > RightHand
#   ├ LeftUpLeg > LeftLeg > LeftFoot
#   └ RightUpLeg > RightLeg > RightFoot
# Skeleton3D 코드 접근
@onready var skeleton: Skeleton3D = (
    $Character/Armature/Skeleton3D
)

func _ready():
    # 본 개수 확인
    print("본 개수: ", skeleton.get_bone_count())

    # 본 이름으로 인덱스 검색
    var head_idx = skeleton.find_bone("Head")
    print("Head 인덱스: ", head_idx)

    # 모든 본 이름 출력
    for i in skeleton.get_bone_count():
        var name = skeleton.get_bone_name(i)
        var parent = skeleton.get_bone_parent(i)
        print("  [%d] %s (부모: %d)" %
            [i, name, parent])

# 특정 본 트랜스폼 제어
func look_at_target(target: Vector3):
    var head_idx = skeleton.find_bone("Head")
    var head_pose = skeleton.get_bone_pose(head_idx)

    # 목표 방향으로 회전 계산
    var direction = (
        target - skeleton.to_global(head_pose.origin)
    ).normalized()
    # 본 포즈 오버라이드
    skeleton.set_bone_pose_rotation(
        head_idx,
        Quaternion(
            head_pose.basis.z, direction
        )
    )

BoneAttachment3D

장착 본에 오브젝트를 부착하여 함께 움직이기

활용 사례

  • 손에 무기(칼, 총, 방패) 장착
  • 머리에 모자, 헬멧, 헤어 악세서리
  • 등에 백팩, 날개, 망토
  • 파티클 이펙트를 특정 본에 부착
  • 카메라를 캐릭터 머리에 부착 (1인칭)
# 씬 트리 구조
# Character (CharacterBody3D)
# └ Armature
#   └ Skeleton3D
#     ├ MeshInstance3D (캐릭터 메시)
#     ├ BoneAttachment3D  [bone: "RightHand"]
#     │ └ Sword (MeshInstance3D)
#     └ BoneAttachment3D  [bone: "Head"]
#       └ Helmet (MeshInstance3D)
# BoneAttachment3D 코드 설정
func setup_weapon_slot():
    var attachment = BoneAttachment3D.new()
    attachment.bone_name = "RightHand"

    # Skeleton3D의 자식으로 추가
    skeleton.add_child(attachment)

    # 무기 메시 추가
    var sword = preload(
        "res://assets/models/sword.glb"
    ).instantiate()
    attachment.add_child(sword)

    # 오프셋 조정 (손에 맞게)
    sword.position = Vector3(0, 0.1, 0)
    sword.rotation_degrees = Vector3(0, 0, -90)

# 무기 교체 시스템
var current_weapon: Node3D = null

func equip_weapon(weapon_scene: PackedScene):
    var slot = $Skeleton3D/RightHandAttachment

    # 기존 무기 제거
    if current_weapon:
        current_weapon.queue_free()

    # 새 무기 장착
    current_weapon = weapon_scene.instantiate()
    slot.add_child(current_weapon)

func unequip_weapon():
    if current_weapon:
        current_weapon.queue_free()
        current_weapon = null
Tip: 에디터에서 BoneAttachment3D를 선택하고 bone_name 속성의 드롭다운에서 원하는 본을 선택하면 됩니다. 오프셋은 부착된 자식 노드의 Transform으로 조절합니다.

func play_animation(anim_name: String):

        return
    # 0.2초 크로스페이드 전환
    anim_player.play(anim_name, 0.2)

func play_slow_motion():
    anim_player.play("attack")
    anim_player.speed_scale = 0.5

# 역재생
func play_reverse():
    anim_player.play_backwards("jump")
# 애니메이션 시그널 활용
func _ready():
    anim_player.animation_finished.connect(
        _on_animation_finished
    )

func _on_animation_finished(anim_name: String):
    match anim_name:
        "attack":
            play_animation("idle")
        "jump":
            play_animation("idle")
        "death":
            queue_free()

AnimationPlayer 3D 활용

본 외에도 다양한 3D 속성을 키프레임으로 애니메이션

애니메이션 가능한 3D 속성

  • 본 트랜스폼 - 위치, 회전, 스케일
  • 노드 Transform - position, rotation
  • 머티리얼 - albedo_color, emission
  • 셰이더 uniform - 커스텀 파라미터
  • 블렌드셰이프 - 얼굴 표정 등
  • 가시성 - visible, transparency
  • 메서드 호출 - 특정 프레임에서 함수 실행
# 코드로 Animation 리소스 생성
func create_door_animation():
    var anim = Animation.new()
    anim.length = 1.0
    anim.loop_mode = Animation.LOOP_NONE

    # 회전 트랙 추가
    var track_idx = anim.add_track(
        Animation.TYPE_VALUE
    )
    anim.track_set_path(
        track_idx,
        "Door:rotation_degrees:y"
    )

    # 키프레임 삽입
    anim.track_insert_key(track_idx, 0.0, 0.0)
    anim.track_insert_key(track_idx, 1.0, -90.0)

    # 보간 모드 설정
    anim.track_set_interpolation_type(
        track_idx,
        Animation.INTERPOLATION_CUBIC
    )

    # AnimationPlayer에 등록
    var player = $AnimationPlayer
    var lib = player.get_animation_library("")
    lib.add_animation("open_door", anim)

# 블렌드셰이프 애니메이션 (표정)
func animate_face():
    var anim_player = $AnimationPlayer
    # 블렌드셰이프 트랙이 임포트 시 자동 생성
    # "Smile", "Blink", "OpenMouth" 등
    anim_player.play("smile_expression")
Tip: 에디터에서 AnimationPlayer를 선택하고 하단 패널의 Animation 탭에서 키 아이콘을 클릭하면 현재 선택된 노드의 속성에 키프레임을 바로 추가할 수 있습니다.

BlendSpace1D / BlendSpace2D

블렌딩 파라미터 기반 자동 애니메이션 보간

BlendSpace1D (속도 기반)

# BlendSpace1D 구성:
# 위치 -1.0: walk_backward
# 위치  0.0: idle
# 위치  0.5: walk
# 위치  1.0: run
#
# 하나의 blend_position 값으로 자동 보간

@onready var anim_tree: AnimationTree = (
    $AnimationTree
)

func _physics_process(delta: float):
    var speed = velocity.length()
    var max_speed = 10.0

    # 속도를 0~1로 정규화
    var blend_pos = clampf(
        speed / max_speed, 0.0, 1.0
    )

    anim_tree.set(
        "parameters/BlendSpace1D/blend_position",
        blend_pos
    )

BlendSpace2D (방향 기반)

# BlendSpace2D 구성 (8방향):
#         walk_forward (0, -1)
#            /      \
# walk_left       walk_right
# (-1, 0)           (1, 0)
#            \      /
#        walk_backward (0, 1)
# 중앙 (0, 0): idle

func _physics_process(delta: float):
    # 로컬 속도 계산 (캐릭터 기준)
    var local_vel = (
        global_transform.basis.inverse()
        * velocity
    )
    var max_speed = 6.0

    # X: 좌우, Y: 전후
    var blend_x = clampf(
        local_vel.x / max_speed, -1.0, 1.0
    )
    var blend_y = clampf(
        -local_vel.z / max_speed, -1.0, 1.0
    )

    anim_tree.set(
        "parameters/BlendSpace2D/blend_position",
        Vector2(blend_x, blend_y)
    )
Tip: BlendSpace2D 에디터에서 삼각형 자동 생성 버튼을 클릭하면 배치된 점들을 기반으로 블렌딩 영역이 자동 구성됩니다.

전환 조건 (Travel / Advance)

StateMachine에서 상태 간 전환을 제어하는 방법

전환 모드

모드 동작
Immediate 즉시 전환 (크로스페이드)
Sync 동기화된 전환 (같은 프레임 비율 유지)
At End 현재 애니메이션 종료 후 전환 
# travel: 경로를 따라 전환
func _physics_process(delta: float):
    var sm: AnimationNodeStateMachinePlayback = (
        anim_tree.get("parameters/playback")
    )

    # 현재 상태 확인
    var current = sm.get_current_node()
    print("현재 상태: ", current)

    # 목표 상태로 이동 (경로 자동 탐색)
    if is_on_floor() and velocity.length() > 0.1:
        sm.travel("walk")
    elif not is_on_floor():
        sm.travel("fall")
    else:
        sm.travel("idle")
# Advance 조건 (자동 전환)
# 에디터에서 전환 선에 조건 설정:
# idle -> walk: advance_condition = "is_moving"
# walk -> idle: advance_condition = "is_idle"
# any -> jump: advance_condition = "is_jumping"

# 조건 파라미터 설정
func update_conditions():
    var speed = velocity.length()
    anim_tree.set(
        "parameters/conditions/is_moving",
        speed > 0.1
    )
    anim_tree.set(
        "parameters/conditions/is_idle",
        speed <= 0.1
    )
    anim_tree.set(
        "parameters/conditions/is_jumping",
        not is_on_floor()
    )

# 전환 시간(xfade) 설정
# 에디터에서 전환 선 클릭 후:
# - Switch Mode: Immediate / Sync / At End
# - Xfade Time: 0.2 (크로스페이드 시간)
# - Advance Condition: 조건 이름
# - Priority: 전환 우선순위
주의: travel()은 정의된 전환 경로가 있어야 작동합니다. 경로가 없으면 전환이 무시됩니다. 에디터에서 상태 간 화살표(전환 선)를 반드시 연결하세요.

루트 모션 (Root Motion)

정밀 이동 애니메이션 데이터로 캐릭터를 실제 이동

루트 모션의 장점

  • 발 미끄러짐 방지 (풋 슬라이딩 제거)
  • 애니메이터 의도대로 정확한 이동 거리
  • 복잡한 이동(구르기, 넘기 등)도 자연스러움
  • 물리 기반 이동과 조합 가능
Tip: Blender에서 루트 모션 애니메이션을 만들 때, Hips 본이 실제로 이동하도록 키프레임을 설정하세요. 제자리 애니메이션과 루트 모션 애니메이션을 구분하여 관리하는 것이 좋습니다. 
# AnimationTree에서 루트 모션 설정
# Inspector > AnimationTree:
#   root_motion_track = "Armature/Skeleton3D:Hips"
# 루트 본(Hips)의 이동을 추출

@onready var anim_tree: AnimationTree = (
    $AnimationTree
)
@onready var body: CharacterBody3D = $".."

func _physics_process(delta: float):
    # 루트 모션에서 이동량 추출
    var root_motion = (
        anim_tree.get_root_motion_position()
    )

    # 캐릭터 방향에 맞게 변환
    var motion = (
        body.global_transform.basis * root_motion
    )

    # CharacterBody3D에 적용
    body.velocity.x = motion.x / delta
    body.velocity.z = motion.z / delta

    # 중력은 별도 처리
    if not body.is_on_floor():
        body.velocity.y -= 9.8 * delta

    body.move_and_slide()

    # 루트 모션의 회전도 적용 가능
    var root_rot = (
        anim_tree.get_root_motion_rotation()
    )
    body.global_transform.basis *= Basis(root_rot)

IK (Inverse Kinematics) 개요

역운동학 목표 지점 기반으로 관절 각도 자동 계산

FK vs IK 비교

항목 FK (정운동학) IK (역운동학)
방향 루트에서 끝으로 끝에서 루트로
입력 각 관절의 회전 값 끝단의 목표 위치
용도 애니메이션 클립 재생 환경 적응, 조준
예시 걷기 애니메이션 경사면에 발 맞춤

IK 활용 사례

  • 발이 경사면/계단에 맞게 위치
  • 손이 벽이나 물체를 짚는 동작
  • 무기 조준 시 상체 회전
  • 머리가 목표를 바라보는 LookAt
# IK의 기본 원리 (2본 체인 예시)
#
# 어깨(A) -- 팔꿈치(B) -- 손(C)
#
# FK: A 회전 -> B 위치 결정 -> B 회전 -> C 위치 결정
# IK: C의 목표 위치 설정 -> B, A의 회전 자동 계산
#
# Godot에서 제공하는 IK 노드:
# - SkeletonIK3D (레거시, 간단한 체인)
# - 커스텀 FABRIK/CCD 구현 (고급)

# 간단한 LookAt IK (머리 회전)
func look_at_ik(target_pos: Vector3):
    var skeleton = $Skeleton3D
    var head_idx = skeleton.find_bone("Head")
    var neck_idx = skeleton.find_bone("Neck")

    # 목표 방향 계산
    var head_global = (
        skeleton.global_transform
        * skeleton.get_bone_global_pose(head_idx)
    )
    var dir = (
        target_pos - head_global.origin
    ).normalized()

    # 회전 적용 (제한 범위 내)
    var look_basis = Basis.looking_at(dir)
    var current = skeleton.get_bone_pose_rotation(
        head_idx
    )
    var target_rot = Quaternion(look_basis)
    # 부드럽게 보간
    var result = current.slerp(target_rot, 0.1)
    skeleton.set_bone_pose_rotation(
        head_idx, result
    )

발 IK (Foot Placement)

# 발 IK: 경사면에 발 맞추기
@onready var left_target: Marker3D = (
    $Skeleton3D/LeftFootTarget
)
@onready var right_target: Marker3D = (
    $Skeleton3D/RightFootTarget
)
@onready var ray_left: RayCast3D = $LeftFootRay
@onready var ray_right: RayCast3D = $RightFootRay

func _physics_process(delta: float):
    update_foot_ik(
        ray_left, left_target, delta
    )
    update_foot_ik(
        ray_right, right_target, delta
    )

func update_foot_ik(
    ray: RayCast3D,
    target: Marker3D,
    delta: float
):
    if ray.is_colliding():
        var hit_point = ray.get_collision_point()
        var hit_normal = ray.get_collision_normal()

        # 목표 위치를 충돌 지점으로
        target.global_position = hit_point
        # 법선 방향으로 발 회전
        var foot_basis = Basis.looking_at(
            hit_normal.cross(
                target.global_transform.basis.x
            ),
            hit_normal
        )
        target.global_transform.basis = (
            target.global_transform.basis.slerp(
                foot_basis, delta * 10.0
            )
        )

스프링 시뮬레이션

# 스프링 기반 무기 흔들림 (Sway)
var spring_position := Vector3.ZERO
var spring_velocity := Vector3.ZERO

@export var stiffness: float = 200.0
@export var damping: float = 15.0

func _process(delta: float):
    # 목표 위치와의 차이
    var target = Vector3.ZERO
    var displacement = spring_position - target

    # 스프링 힘 계산
    var spring_force = (
        -stiffness * displacement
        - damping * spring_velocity
    )

    # 적분
    spring_velocity += spring_force * delta
    spring_position += spring_velocity * delta

    # 무기에 적용
    $WeaponPivot.position = spring_position

# 마우스 이동 시 흔들림 추가
func _input(event: InputEvent):
    if event is InputEventMouseMotion:
        var sway = Vector3(
            -event.relative.x * 0.002,
            -event.relative.y * 0.002,
            0
        )
        spring_velocity += sway

# 사인파 기반 숨쉬기 (Breathing)
var breath_time: float = 0.0

func _process(delta: float):
    breath_time += delta
    var breath = sin(breath_time * 1.5) * 0.02
    $Camera3D.position.y = 1.6 + breath

# 노이즈 기반 카메라 흔들림
var shake_intensity: float = 0.0
var shake_decay: float = 5.0
var noise := FastNoiseLite.new()

func _ready():
    noise.noise_type = FastNoiseLite.TYPE_PERLIN
    noise.frequency = 1.0

func camera_shake(intensity: float):
    shake_intensity = intensity

func _process(delta: float):
    if shake_intensity > 0.01:
        var t = Time.get_ticks_msec() * 0.1
        var offset = Vector3(
            noise.get_noise_1d(t) * shake_intensity,
            noise.get_noise_1d(t + 100) * shake_intensity,
            0
        )
        $Camera3D.position += offset
        shake_intensity = lerp(
            shake_intensity, 0.0,
            shake_decay * delta
        )

종합 AnimationTree + IK + Tween 통합 시스템

# character_animation_controller.gd
# 완성된 3D 캐릭터 애니메이션 컨트롤러
extends Node3D

@onready var body: CharacterBody3D = $".."
@onready var anim_tree: AnimationTree = $AnimationTree
@onready var state_machine: AnimationNodeStateMachinePlayback = (
    anim_tree.get("parameters/playback")
)
@onready var skeleton: Skeleton3D = $Armature/Skeleton3D

# 상태 관리
enum State { LOCOMOTION, ATTACK, DODGE, HIT, DEATH }
var current_state: State = State.LOCOMOTION

func _ready():
    anim_tree.active = true

func _physics_process(delta: float):
    match current_state:
        State.LOCOMOTION:
            update_locomotion(delta)
        State.ATTACK:
            pass  # AnimationTree가 처리
        State.HIT:
            pass  # 피격 애니메이션 재생 중

# 이동 블렌딩 (BlendSpace2D)
func update_locomotion(delta: float):
    var vel = body.global_transform.basis.inverse() * body.velocity
    var max_speed = 6.0
    var blend = Vector2(
        clampf(vel.x / max_speed, -1.0, 1.0),
        clampf(-vel.z / max_speed, -1.0, 1.0)
    )
    anim_tree.set("parameters/Locomotion/blend_position", blend)

# 공격 시작
func play_attack(attack_name: String):
    current_state = State.ATTACK
    # OneShot 노드로 공격 삽입
    anim_tree.set("parameters/AttackOneShot/request", AnimationNodeOneShot.ONE_SHOT_REQUEST_FIRE)
    await anim_tree.animation_finished
    current_state = State.LOCOMOTION

# 피격 효과
func on_hit(direction: Vector3):
    current_state = State.HIT
    state_machine.travel("hit")
    flash_damage()
    await get_tree().create_timer(0.5).timeout
    current_state = State.LOCOMOTION

func flash_damage():
    var mat = $Armature/Skeleton3D/MeshInstance3D.get_surface_override_material(0)
    if mat:
        var tween = create_tween()
        tween.tween_property(mat, "emission_energy_multiplier", 3.0, 0.05)
        tween.tween_property(mat, "emission_energy_multiplier", 0.0, 0.3)
Tip: 각 기능(이동 블렌딩, 공격, 피격, IK)을 별도 메서드로 분리하면 유지보수와 확장이 쉽습니다. 상태 패턴과 결합하면 더욱 견고한 시스템이 됩니다.
