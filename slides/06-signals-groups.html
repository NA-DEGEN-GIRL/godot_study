시그널이란?

Observer Pattern (옵저버 패턴)

시그널은 이벤트 발생을 알리는 메시지 시스템입니다. 보내는 쪽(Emitter)은 받는 쪽(Observer)을 알 필요 없이, 구독한 모든 노드에게 자동으로 전달됩니다.

직접 호출 (강한 결합)

# 나쁜 예: 보내는 쪽이 받는 쪽을 알아야 함
func die():
    $"../UI/HealthBar".update(0)
    $"../AudioManager".play("death")
    $"../GameManager".on_player_dead()
    # 노드가 바뀌면 전부 수정해야 함!

시그널 (느슨한 결합)

# 좋은 예: 시그널로 알림만 보냄
signal died

func die():
    died.emit()
    # 누가 듣든 상관없음!
    # UI, Audio, GM이 각자 연결
원칙: 자식 노드에는 직접 함수를 호출하고, 부모나 형제 노드에는 시그널을 사용하세요. 이것이 Godot의 권장 패턴입니다.

내장 시그널 사용

Godot의 모든 노드는 미리 정의된 시그널을 가지고 있습니다

노드시그널발생 시점
Buttonpressed버튼 클릭 시
Area2Dbody_entered바디가 영역 진입 시
Timertimeout타이머 완료 시
AnimationPlayeranimation_finished애니메이션 종료 시
CharacterBody2D-(내장 시그널 없음, 커스텀 사용)
Nodeready, tree_entered초기화/트리 진입 시
확인 방법: 에디터에서 노드 선택 후 우측 Node 탭 (인스펙터 옆)을 클릭하면 해당 노드의 모든 시그널 목록을 볼 수 있습니다.

connect() 메서드

Godot 4 문법 시그널 객체에 직접 .connect() 호출

# Godot 4 시그널 연결 기본 문법
# 시그널.connect(콜백함수)

func _ready():
    # 1) 기본 연결
    $Button.pressed.connect(_on_button_pressed)

    # 2) 람다 함수로 간단하게
    $Button.pressed.connect(func(): print("클릭!"))

    # 3) 인자 바인딩
    $Button.pressed.connect(_on_item_pressed.bind("sword"))

func _on_button_pressed():
    print("버튼이 눌렸습니다!")

func _on_item_pressed(item_name: String):
    print(item_name, " 선택됨!")
# 연결 플래그 (flags)
func _ready():
    # 한 번만 실행되고 자동 해제
    $Timer.timeout.connect(_on_timeout, CONNECT_ONE_SHOT)

    # 지연 호출 (현재 프레임 끝에 실행)
    $Area2D.body_entered.connect(_on_enter, CONNECT_DEFERRED)
Godot 3 vs 4: Godot 3의 connect("signal_name", self, "method") 문법은 더 이상 사용하지 않습니다. Godot 4에서는 signal.connect(callable) 형식을 사용합니다.

에디터에서 시그널 연결

에디터 연결 순서

  1. 시그널을 보낼 노드 선택 (예: Button)
  2. 우측 Node 탭 클릭
  3. 원하는 시그널 더블클릭 (예: pressed)
  4. 연결할 대상 노드 선택
  5. 메서드 이름 확인 후 Connect 클릭

에디터 연결 특징

  • 스크립트에 초록색 아이콘 표시
  • 자동으로 _on_노드명_시그널명 생성
  • .tscn 파일에 연결 정보 저장
  • 씬 내부 노드끼리만 가능

자동 생성되는 코드

# 에디터가 자동 생성한 콜백
func _on_button_pressed():
    pass  # 여기에 로직 추가

func _on_timer_timeout():
    pass

func _on_area_2d_body_entered(body):
    pass
권장: 씬 내부의 고정 노드는 에디터 연결, 런타임에 동적 생성되는 노드는 코드 연결을 사용하세요.

코드에서 시그널 연결

# 기본: _ready에서 연결
func _ready():
    $StartButton.pressed.connect(_on_start)
    $QuitButton.pressed.connect(_on_quit)
    $Timer.timeout.connect(_on_wave_timeout)

func _on_start():
    get_tree().change_scene_to_file("res://game.tscn")

func _on_quit():
    get_tree().quit()
# 동적 생성 노드에 시그널 연결
func spawn_enemy():
    var enemy = enemy_scene.instantiate()
    add_child(enemy)
    # 생성 즉시 시그널 연결
    enemy.died.connect(_on_enemy_died.bind(enemy))

func _on_enemy_died(enemy: Node2D):
    score += enemy.point_value
    enemy.queue_free()
# 여러 버튼에 반복문으로 연결
func _ready():
    for button in $ButtonContainer.get_children():
        if button is Button:
            button.pressed.connect(_on_menu_button.bind(button))

func _on_menu_button(button: Button):
    print(button.name, " 클릭됨")

시그널 해제 (disconnect)

# 시그널 연결 해제
func _ready():
    $Button.pressed.connect(_on_pressed)

func disable_button():
    # 연결 상태 확인 후 해제
    if $Button.pressed.is_connected(_on_pressed):
        $Button.pressed.disconnect(_on_pressed)

func _on_pressed():
    print("눌렸습니다")

자동 해제되는 경우

  • 수신 노드가 queue_free()로 제거될 때
  • CONNECT_ONE_SHOT 플래그 사용 시
  • 발신 노드가 제거될 때

수동 해제가 필요한 경우

  • 조건부 활성화/비활성화
  • 콜백 함수 변경
  • 중복 연결 방지
  • 일시적 이벤트 리스너
주의: 람다 함수로 연결하면 참조를 저장하지 않는 한 disconnect가 불가능합니다. 해제가 필요하면 반드시 이름 있는 함수를 사용하세요.

커스텀 시그널 선언

signal 키워드 게임 로직에 맞는 나만의 시그널 만들기

# 커스텀 시그널 선언 (스크립트 상단)
extends CharacterBody2D

signal health_changed     # 매개변수 없는 시그널
signal died               # 사망 시그널
signal item_collected     # 아이템 수집 시그널

var health := 100

func take_damage(amount: int):
    health -= amount
    health_changed.emit()    # 시그널 발생!

    if health <= 0:
        died.emit()          # 사망 시그널 발생!

func collect_item(item):
    item_collected.emit()    # 아이템 수집 시그널!
# 다른 노드에서 연결
# ui_manager.gd
func _ready():
    var player = get_node("/root/Game/Player")
    player.health_changed.connect(_on_health_changed)
    player.died.connect(_on_player_died)

func _on_health_changed():
    print("체력이 변했습니다!")

func _on_player_died():
    show_game_over_screen()

extends CharacterBody2D

signal health_changed(new_health: int, max_health: int)

signal level_up(new_level: int)
signal item_picked(item_name: String, quantity: int)

var level := 1

    health = max(0, health - amount)
    # 매개변수와 함께 emit
    damage_taken.emit(amount, source)
    health_changed.emit(health, max_health)

func gain_exp(amount: int):
    # ... 경험치 로직 ...
    if should_level_up:
        level += 1
        level_up.emit(level)

    $HealthLabel.text = "%d / %d" % [current, maximum]

func _on_damage_taken(amount: int, source: String):

시그널 발생 (emit)

Godot 4 방식 (권장)

# 시그널 객체에서 직접 호출
signal scored(points: int)
signal game_over

func add_score(points: int):
    scored.emit(points)

func die():
    game_over.emit()

Godot 3 호환 방식

# 문자열 기반 (비추천)
func add_score(points: int):
    emit_signal("scored", points)

func die():
    emit_signal("game_over")
# 오타 시 런타임 에러 위험!
# await로 시그널 대기하기 (매우 유용!)
func play_cutscene():
    $AnimationPlayer.play("intro")
    # 애니메이션이 끝날 때까지 기다림
    await $AnimationPlayer.animation_finished
    print("애니메이션 완료! 다음 로직 실행")

    $Timer.start(2.0)
    await $Timer.timeout
    print("2초 후 실행!")

    start_gameplay()
await 패턴: await 시그널은 비동기 프로그래밍의 핵심입니다. 컷신, 대화, 타이머 대기 등에 코드를 순차적으로 작성할 수 있어 가독성이 크게 향상됩니다.

그룹이란?

Group = 노드 태그 시스템

노드에 문자열 태그를 붙여 분류하고, 같은 그룹의 노드를 한 번에 검색하거나 일괄 처리할 수 있습니다.

그룹 활용 예시

  • "enemy" - 모든 적 노드
  • "player" - 플레이어
  • "destructible" - 파괴 가능 물체
  • "save_target" - 저장 대상
  • "pausable" - 일시정지 대상

특징

  • 노드 하나가 여러 그룹에 속할 수 있음
  • 문자열 기반으로 유연함
  • 에디터 또는 코드로 설정 가능
  • SceneTree를 통해 전역 검색
  • 그룹 함수 일괄 호출 가능
시그널 vs 그룹: 시그널은 1:N 이벤트 전달, 그룹은 N개 노드 일괄 처리입니다. 목적이 다르므로 적절히 조합하세요.

add_to_group / is_in_group

# 그룹 추가/제거/확인
extends CharacterBody2D

func _ready():
    # 그룹에 추가
    add_to_group("player")
    add_to_group("damageable")
    add_to_group("save_target")

func check_groups():
    # 그룹 소속 확인
    if is_in_group("player"):
        print("플레이어입니다!")

func change_team():
    # 그룹에서 제거
    remove_from_group("ally")
    # 다른 그룹에 추가
    add_to_group("enemy")
# 충돌 시 그룹으로 대상 판별
func _on_body_entered(body: Node2D):
    if body.is_in_group("player"):
        deal_damage_to_player(body)
    elif body.is_in_group("enemy"):
        deal_damage_to_enemy(body)
    elif body.is_in_group("destructible"):
        body.destroy()
에디터 설정: 노드 선택 후 Node 탭 > Groups에서 그룹을 추가하면, 씬 파일에 저장되어 매번 코드로 추가할 필요가 없습니다.

get_tree().call_group()

일괄 호출 그룹 내 모든 노드의 함수를 한 번에 호출

# call_group: 그룹 내 모든 노드의 메서드 호출
func kill_all_enemies():
    get_tree().call_group("enemy", "die")

func pause_all_enemies():
    get_tree().call_group("enemy", "set_paused", true)

func heal_all_allies():
    get_tree().call_group("ally", "heal", 50)
# get_nodes_in_group: 그룹의 모든 노드를 배열로 가져오기
func count_enemies() -> int:
    var enemies = get_tree().get_nodes_in_group("enemy")
    return enemies.size()

func get_closest_enemy(pos: Vector2) -> Node2D:
    var enemies = get_tree().get_nodes_in_group("enemy")
    var closest: Node2D = null
    var min_dist := INF

    for enemy in enemies:
        var dist = pos.distance_to(enemy.global_position)
        if dist < min_dist:
            min_dist = dist
            closest = enemy

    return closest

그룹으로 일괄 처리

게임 저장 시스템

# save_manager.gd
func save_game():
    var save_data := {}
    var targets = get_tree() \
        .get_nodes_in_group("save_target")

    for node in targets:
        if node.has_method("get_save_data"):
            var key = node.get_path()
            save_data[key] = \
                node.get_save_data()

    # 파일에 저장
    var file = FileAccess.open(
        "user://save.json",
        FileAccess.WRITE
    )
    file.store_string(
        JSON.stringify(save_data)
    )

폭발 범위 처리

# explosion.gd
func explode():
    var targets = get_tree() \
        .get_nodes_in_group("destructible")

    for target in targets:
        var dist = global_position \
            .distance_to(
                target.global_position
            )
        if dist <= explosion_radius:
            # 거리에 따른 데미지
            var ratio = 1.0 - (
                dist / explosion_radius
            )
            var dmg = int(
                max_damage * ratio
            )
            target.take_damage(dmg)

시그널 vs 그룹 비교

구분시그널 (Signal)그룹 (Group)
패턴Observer (관찰자)Tag (태그/분류)
방향발신자 1 : 수신자 N호출자 1 : 대상 N
결합도느슨 (발신자가 수신자 모름)보통 (호출자가 메서드명 알아야 함)
연결수신자가 발신자에 connectget_tree().call_group()
시점이벤트 발생 시필요할 때 직접 호출
예시체력 변화, 사망, 점수 획득적 전체 제거, 저장, 일시정지

조합 패턴

시그널로 이벤트 감지 후, 그룹으로 일괄 처리 - 예: 보스 사망 시그널 수신 후 call_group("enemy", "retreat")으로 모든 적 후퇴 명령

시그널 버스 패턴 (Autoload)

전역 이벤트 시스템 Autoload 싱글톤으로 어디서든 시그널 주고받기

# event_bus.gd (Autoload로 등록: Project > Project Settings > Autoload)
extends Node

# 전역 시그널 선언
signal game_started

signal score_changed(new_score: int)
signal player_died
signal level_completed(level_num: int)

주의: 시그널 버스를 남용하면 어디서 시그널이 발생하는지 추적이 어려워집니다. 정말 전역적인 이벤트만 버스에 넣고, 지역적 이벤트는 일반 시그널을 사용하세요.

실전: 체력 시스템 예제

종합 예제 시그널 + 그룹 활용 체력 관리

# health_component.gd - 재사용 가능한 체력 컴포넌트
extends Node
class_name HealthComponent

signal health_changed(current: int, maximum: int)
signal damage_taken(amount: int)
signal healed(amount: int)
signal died

@export var max_health: int = 100
var current_health: int

func _ready():
    current_health = max_health
    get_parent().add_to_group("damageable")

func take_damage(amount: int):
    if current_health <= 0:
        return
    current_health = max(0, current_health - amount)
    damage_taken.emit(amount)
    health_changed.emit(current_health, max_health)
    if current_health <= 0:
        died.emit()

func heal(amount: int):
    var old = current_health
    current_health = min(max_health, current_health + amount)
    var actual = current_health - old
    if actual > 0:
        healed.emit(actual)
        health_changed.emit(current_health, max_health)
# health_bar_ui.gd - UI가 시그널에 반응
extends ProgressBar

@export var health_component: HealthComponent

func _ready():
    if health_component:
        health_component.health_changed.connect(_on_health_changed)
        health_component.died.connect(_on_died)
        value = 100  # 초기값

func _on_health_changed(current: int, maximum: int):
    value = float(current) / maximum * 100

func _on_died():
    modulate = Color.RED

Chapter 06 정리

핵심 요약

  • 시그널: 이벤트 발생을 알리는 옵저버 패턴 (느슨한 결합)
  • connect/emit: 시그널 연결과 발생의 핵심 메서드
  • 커스텀 시그널: signal 키워드로 게임 고유 이벤트 정의
  • 그룹: 노드를 태그로 분류하여 일괄 처리
개념핵심 내용
signal 키워드커스텀 시그널 선언
.connect(callable)시그널에 콜백 함수 연결
.emit(args)시그널 발생 (데이터 전달)
await signal시그널 대기 (비동기 흐름 제어)
add_to_group()런타임 그룹 추가
call_group()그룹 내 함수 일괄 호출
시그널 버스Autoload 전역 이벤트 시스템
